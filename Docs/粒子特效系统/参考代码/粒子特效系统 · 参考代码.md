## 1. `Foundation/Pool.cs`（改造）

```csharp
using System;
using System.Collections.Generic;
using UnityEngine;

namespace DeepseaOil.Foundation
{
    public enum PoolOverflowPolicy
    {
        /// <summary>池空时现场创建，LogWarning</summary>
        CreateAndWarn,
        /// <summary>池空时丢弃本次请求，TryGet 返回 false</summary>
        DropSilently,
        /// <summary>池空时抛 InvalidOperationException</summary>
        Throw,
    }

    public readonly struct PoolStats
    {
        public readonly int Active;         // 当前借出数
        public readonly int Idle;           // 池中待用数
        public readonly int Peak;           // 历史峰值 Active
        public readonly int TotalCreated;   // 累计创建
        public readonly int TotalDropped;   // 累计丢弃

        public PoolStats(int active, int idle, int peak, int totalCreated, int totalDropped)
        {
            Active = active; Idle = idle; Peak = peak;
            TotalCreated = totalCreated; TotalDropped = totalDropped;
        }
    }

    /// <summary>
    /// 通用对象池。
    /// 语义：
    ///   - TryGet(out T)：池空时按策略处理，返回是否成功借出
    ///   - Release(T)：归还
    ///   - Prewarm(n)：预创建 n 个对象，不触发 onGet/onRelease
    ///   - Clear()：清空 idle 部分；提供 onDestroy 时逐个销毁
    ///   - Dispose()：同 Clear + 标记不可用
    /// </summary>
    public sealed class Pool<T> : IDisposable where T : class
    {
        private readonly string _name;
        private readonly Func<T> _factory;
        private readonly Action<T> _onGet;
        private readonly Action<T> _onRelease;
        private readonly Action<T> _onDestroy;
        private readonly int _maxSize;
        private readonly PoolOverflowPolicy _overflowPolicy;
        private readonly Stack<T> _idle = new();

        private int _active;
        private int _peak;
        private int _totalCreated;
        private int _totalDropped;
        private bool _disposed;

        public string Name => _name;
        public int MaxSize => _maxSize;
        public int ActiveCount => _active;
        public int IdleCount => _idle.Count;

        public Pool(
            string name,
            Func<T> factory,
            int maxSize = int.MaxValue,
            PoolOverflowPolicy overflowPolicy = PoolOverflowPolicy.CreateAndWarn,
            Action<T> onGet = null,
            Action<T> onRelease = null,
            Action<T> onDestroy = null)
        {
            _name = string.IsNullOrEmpty(name) ? typeof(T).Name : name;
            _factory = factory ?? throw new ArgumentNullException(nameof(factory));
            if (maxSize <= 0) throw new ArgumentOutOfRangeException(nameof(maxSize));
            _maxSize = maxSize;
            _overflowPolicy = overflowPolicy;
            _onGet = onGet;
            _onRelease = onRelease;
            _onDestroy = onDestroy;
        }

        public bool TryGet(out T obj)
        {
            ThrowIfDisposed();
            obj = null;

            if (_idle.Count > 0)
            {
                obj = _idle.Pop();
                _active++;
                if (_active > _peak) _peak = _active;
                _onGet?.Invoke(obj);
                return true;
            }

            switch (_overflowPolicy)
            {
                case PoolOverflowPolicy.CreateAndWarn:
                    Debug.LogWarning($"[Pool<{_name}>] 池空，现场创建。考虑 Prewarm 或提高 MaxSize。");
                    obj = CreateNew();
                    _active++;
                    if (_active > _peak) _peak = _active;
                    _onGet?.Invoke(obj);
                    return true;

                case PoolOverflowPolicy.DropSilently:
                    _totalDropped++;
                    return false;

                case PoolOverflowPolicy.Throw:
                    _totalDropped++;
                    throw new InvalidOperationException($"[Pool<{_name}>] 池空且策略为 Throw。");

                default:
                    return false;
            }
        }

        /// <summary>池空且策略为 DropSilently 时返回 null，调用方需判空。</summary>
        public T Get() => TryGet(out var obj) ? obj : null;

        public void Release(T obj)
        {
            ThrowIfDisposed();
            if (obj == null) throw new ArgumentNullException(nameof(obj));

            _onRelease?.Invoke(obj);
            _active--;
            if (_active < 0)
            {
                Debug.LogError($"[Pool<{_name}>] Active 计数小于 0，Release 次数多于 Get。");
                _active = 0;
            }

            if (_idle.Count >= _maxSize)
            {
                // 超出上限：直接销毁
                _onDestroy?.Invoke(obj);
                return;
            }

            _idle.Push(obj);
        }

        /// <summary>预创建 count 个对象入池。不触发 onGet / onRelease。</summary>
        public void Prewarm(int count)
        {
            ThrowIfDisposed();
            if (count < 0) throw new ArgumentOutOfRangeException(nameof(count));

            int toCreate = Math.Min(count, _maxSize - _idle.Count);
            for (int i = 0; i < toCreate; i++)
            {
                _idle.Push(CreateNew());
            }
        }

        /// <summary>清空 idle 部分。不影响已借出对象。</summary>
        public void Clear()
        {
            if (_onDestroy != null)
            {
                foreach (var obj in _idle) _onDestroy(obj);
            }
            _idle.Clear();
        }

        public PoolStats GetStats()
            => new PoolStats(_active, _idle.Count, _peak, _totalCreated, _totalDropped);

        public void Dispose()
        {
            if (_disposed) return;
            Clear();
            _disposed = true;
        }

        private T CreateNew()
        {
            _totalCreated++;
            return _factory();
        }

        private void ThrowIfDisposed()
        {
            if (_disposed) throw new ObjectDisposedException($"[Pool<{_name}>]");
        }
    }
}
```

```csharp
using UnityEngine;

namespace DeepseaOil.Presentation.Effects
{
    /// <summary>
    /// 一次特效播放的参数。纯数据，不含任何逻辑。
    /// Intensity 语义由 Driver 解释：ScreenShake 用于取最大值合并，HitSpark 用于缩放粒子量。
    /// </summary>
    public struct EffectContext
    {
        public Vector2   Position;
        public Vector2   Direction;
        public float     Scale;       // 默认 1
        public float     Intensity;   // 0~1
        public Transform Follow;      // 可空

        public static EffectContext At(Vector2 position)
            => new EffectContext
            {
                Position = position,
                Direction = Vector2.up,
                Scale = 1f,
                Intensity = 1f,
                Follow = null,
            };

        public static EffectContext At(Vector2 position, Vector2 direction)
            => new EffectContext
            {
                Position = position,
                Direction = direction,
                Scale = 1f,
                Intensity = 1f,
                Follow = null,
            };

        public static EffectContext OnTarget(Transform target)
            => new EffectContext
            {
                Position = target != null ? (Vector2)target.position : Vector2.zero,
                Direction = Vector2.up,
                Scale = 1f,
                Intensity = 1f,
                Follow = target,
            };
    }
}
```

## 4. `Presentation/Effects/EffectHandle.cs`

```csharp
using System;

namespace DeepseaOil.Presentation.Effects
{
    /// <summary>
    /// 一次播放的句柄。带 Generation，防止 Id 复用后旧 handle 误停新实例。
    /// </summary>
    public readonly struct EffectHandle : IEquatable<EffectHandle>
    {
        internal readonly int Id;
        internal readonly int Generation;
        internal readonly IEffectDriver Driver;  // 哪个 Driver 返回的

        internal EffectHandle(int id, int generation, IEffectDriver driver)
        {
            Id = id;
            Generation = generation;
            Driver = driver;
        }

        public bool IsValid => Id != 0 && Driver != null;

        public static EffectHandle None => default;

        public bool Equals(EffectHandle other)
            => Id == other.Id
            && Generation == other.Generation
            && ReferenceEquals(Driver, other.Driver);

        public override bool Equals(object obj)
            => obj is EffectHandle h && Equals(h);

        public override int GetHashCode() => HashCode.Combine(Id, Generation);

        public override string ToString()
            => IsValid ? $"Effect#{Id}g{Generation}" : "None";

        public static bool operator ==(EffectHandle a, EffectHandle b) => a.Equals(b);
        public static bool operator !=(EffectHandle a, EffectHandle b) => !a.Equals(b);
    }
}
```

## 7. `Presentation/Effects/EffectModule.cs`

```csharp
using System.Collections.Generic;
using UnityEngine;

namespace DeepseaOil.Presentation.Effects
{
    /// <summary>
    /// 特效模块。纯工具，谁调都行。不认识 EventBus，不认识 Logic。
    /// 调用方：表现层的组件（EnemyController.OnHit 等）。
    /// 生命周期由 GameRoot 驱动。
    /// 线程约束：仅主线程调用。
    /// </summary>
    public static class EffectModule
    {
        private static readonly Dictionary<EffectId, IEffectDriver> _drivers = new();
        private static bool _initialized;

        public static bool IsInitialized => _initialized;

        // ─────────────────────────────────────────────
        // 生命周期
        // ─────────────────────────────────────────────

        public static void Init()
        {
            if (_initialized)
            {
                Debug.LogError("[EffectModule] Init called twice");
                return;
            }
            _drivers.Clear();
            _initialized = true;
        }

        public static void Tick(float dt)
        {
            if (!_initialized) return;
            foreach (var kv in _drivers) kv.Value.Tick(dt);
        }

        public static void Dispose()
        {
            if (!_initialized) return;
            CleanAll();
            _drivers.Clear();
            _initialized = false;
        }

        // ─────────────────────────────────────────────
        // 对外接口
        // ─────────────────────────────────────────────

        public static EffectHandle Play(EffectId id, in EffectContext ctx)
        {
            if (!_initialized)
            {
                Debug.LogError("[EffectModule] Play before Init");
                return EffectHandle.None;
            }
            if (id == EffectId.None) return EffectHandle.None;

            if (!_drivers.TryGetValue(id, out var driver))
            {
                Debug.LogError($"[EffectModule] EffectId {id} 未注册");
                return EffectHandle.None;
            }
            return driver.Play(id, in ctx);
        }

        public static void Stop(EffectHandle handle)
        {
            if (!_initialized) return;
            if (!handle.IsValid) return;
            handle.Driver.Stop(handle);
        }

        /// <summary>清空所有活跃实例，全部归还池。切场景时调用。</summary>
        public static void CleanAll()
        {
            if (!_initialized) return;
            foreach (var kv in _drivers) kv.Value.CleanAll();
        }

        public static EffectStats GetStats()
        {
            var stats = new EffectStats { DriverCount = _drivers.Count };
            foreach (var kv in _drivers)
            {
                stats.ActiveInstances += kv.Value.ActiveInstanceCount;
                stats.PooledObjects += kv.Value.PooledObjectCount;
            }
            return stats;
        }

        // ─────────────────────────────────────────────
        // 注册（内部，Driver 由装配处构造后注册）
        // ─────────────────────────────────────────────

        internal static void Register(EffectId id, IEffectDriver driver)
        {
            if (id == EffectId.None)
            {
                Debug.LogError("[EffectModule] 不能注册 None");
                return;
            }
            if (driver == null)
            {
                Debug.LogError($"[EffectModule] Driver for {id} is null");
                return;
            }
            if (_drivers.ContainsKey(id))
            {
                Debug.LogError($"[EffectModule] {id} 重复注册，后覆盖前");
            }
            _drivers[id] = driver;
        }
    }
}
```

## 8. `Presentation/Effects/Drivers/ParticleDriver.cs`

```csharp
using System.Collections.Generic;
using UnityEngine;
using DeepseaOil.Foundation;

namespace DeepseaOil.Presentation.Effects.Drivers
{
    /// <summary>
    /// 粒子驱动器。管理一个 EffectId 对应的粒子 prefab 的播放。
    /// 支持单例型（合并）与多实例型（每次都新建）。
    /// 内部池化，池满时丢弃新请求。
    /// </summary>
    internal sealed class ParticleDriver : IEffectDriver
    {
        private readonly GameObject _prefab;
        private readonly bool _isSingleton;
        private readonly Pool<GameObject> _pool;
        private readonly Transform _root;
        private readonly Dictionary<int, ParticleInstance> _active = new();

        private int _nextInstanceId;

        public int ActiveInstanceCount => _active.Count;
        public int PooledObjectCount => _pool.IdleCount;

        public ParticleDriver(GameObject prefab, Transform root, bool isSingleton, int maxSize, int prewarm = 0)
        {
            _prefab = prefab;
            _root = root;
            _isSingleton = isSingleton;

            _pool = new Pool<GameObject>(
                name: $"ParticleDriver[{prefab.name}]",
                factory: () =>
                {
                    var go = Object.Instantiate(prefab, root);
                    go.SetActive(false);
                    return go;
                },
                maxSize: maxSize,
                overflowPolicy: PoolOverflowPolicy.DropSilently,
                onGet: go => go.SetActive(true),
                onRelease: go => go.SetActive(false),
                onDestroy: go => Object.Destroy(go));

            if (prewarm > 0) _pool.Prewarm(prewarm);
        }

        public EffectHandle Play(EffectId id, in EffectContext ctx)
        {
            if (_prefab == null)
            {
                Debug.LogError($"[ParticleDriver] prefab is null for {id}");
                return EffectHandle.None;
            }

            // 单例：已在播 → 合并
            if (_isSingleton && TryMergeSingleton(in ctx, out var mergedHandle))
            {
                return mergedHandle;
            }

            if (!_pool.TryGet(out var go))
            {
                Debug.LogWarning($"[ParticleDriver] pool full for {id}, dropped.");
                return EffectHandle.None;
            }

            var ps = go.GetComponent<ParticleSystem>();
            if (ps == null)
            {
                Debug.LogError($"[ParticleDriver] prefab has no ParticleSystem");
                _pool.Release(go);
                return EffectHandle.None;
            }

            // 摆放与播放
            go.transform.position = ctx.Position;
            go.transform.localScale = Vector3.one * Mathf.Max(0.01f, ctx.Scale);
            ps.Clear(true);
            ps.Play(true);

            // 记录实例
            var inst = new ParticleInstance
            {
                Id = ++_nextInstanceId,
                GameObject = go,
                ParticleSystem = ps,
                FollowRequested = ctx.Follow != null,
                Follow = ctx.Follow,
                Elapsed = 0f,
                Duration = ps.main.duration + ps.main.startLifetime.constantMax,
                IsSingleton = _isSingleton,
                Intensity = ctx.Intensity,
            };
            _active[inst.Id] = inst;

            return new EffectHandle(inst.Id, 0, this);
        }

        private bool TryMergeSingleton(in EffectContext ctx, out EffectHandle handle)
        {
            foreach (var kv in _active)
            {
                if (kv.Value.IsSingleton)
                {
                    kv.Value.Intensity = Mathf.Max(kv.Value.Intensity, ctx.Intensity);
                    handle = new EffectHandle(kv.Key, 0, this);
                    return true;
                }
            }
            handle = EffectHandle.None;
            return false;
        }

        public void Stop(EffectHandle handle)
        {
            if (!handle.IsValid) return;
            if (!_active.TryGetValue(handle.Id, out var inst)) return;
            ReleaseInstance(inst);
        }

        public void CleanAll()
        {
            var ids = new List<int>(_active.Keys);
            foreach (var id in ids)
            {
                if (_active.TryGetValue(id, out var inst)) ReleaseInstance(inst);
            }
        }

        public void Tick(float dt)
        {
            if (_active.Count == 0) return;

            List<int> toRemove = null;

            foreach (var kv in _active)
            {
                var inst = kv.Value;
                inst.Elapsed += dt;

                // Follow 目标被销毁 → 停止
                if (inst.FollowRequested && inst.Follow == null)
                {
                    (toRemove ??= new List<int>()).Add(inst.Id);
                    continue;
                }
                if (inst.FollowRequested)
                {
                    inst.GameObject.transform.position = inst.Follow.position;
                }

                // 播完
                if (inst.Elapsed >= inst.Duration)
                {
                    (toRemove ??= new List<int>()).Add(inst.Id);
                }
            }

            if (toRemove != null)
            {
                foreach (var id in toRemove)
                {
                    if (_active.TryGetValue(id, out var inst)) ReleaseInstance(inst);
                }
            }
        }

        private void ReleaseInstance(ParticleInstance inst)
        {
            _active.Remove(inst.Id);

            if (inst.ParticleSystem != null)
            {
                inst.ParticleSystem.Stop(true, ParticleSystemStopBehavior.StopEmittingAndClear);
            }
            if (inst.GameObject != null)
            {
                _pool.Release(inst.GameObject);
            }
        }

        private sealed class ParticleInstance
        {
            public int Id;
            public GameObject GameObject;
            public ParticleSystem ParticleSystem;
            public bool FollowRequested;
            public Transform Follow;
            public float Elapsed;
            public float Duration;
            public bool IsSingleton;
            public float Intensity;
        }
    }
}
```

## 装配示意（不需你接线，仅说明）

```
EffectModule.Init();

// 例：HitSpark 多实例，EnemyFlashWhite 多实例，ScreenShake 单例

    new ParticleDriver(hitSparkPrefab, effectsRoot, isSingleton: false, maxSize: 32, prewarm: 8));

    new ParticleDriver(flashPrefab, effectsRoot, isSingleton: false, maxSize: 16, prewarm: 4));

    new ParticleDriver(shakePrefab, effectsRoot, isSingleton: true, maxSize: 1, prewarm: 1));

    new ParticleDriver(mudPrefab, effectsRoot, isSingleton: false, maxSize: 16, prewarm: 4));
```
