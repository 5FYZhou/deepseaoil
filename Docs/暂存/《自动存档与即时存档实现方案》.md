# 《自动存档与即时存档实现方案》
**版本**：v1.0  
**适用范围**：Unity 2022.3 LTS + URP 14.0.12 (2D Renderer)  
**设计目标**：实现“备战期自动存盘 + 战斗中全现场手动存盘 + 槽位带 URP 后处理缩略图展示 + 故障安全原子落盘”。

---

## 1. 架构总览与核心设计原则

### 1.1 第一性原则与反过度设计
1. **备忘录汇流（Pull-based Memento），彻底摒弃集中黑板**：
   不设立持有脏数据的全局单例“黑板”，各子系统（`PlayerStats`、`WaveLogic`、`GridLogic`、`CombatDirector`）仍是自身业务状态的唯一真源。由 `SaveService` 在存盘时刻统一拉取各子系统的只读 DTO，组装为不可变快照 `GameSaveData`。
2. **纯数据序列化（0 引擎耦合）**：
   持久化载荷完全由基础类型（`int`, `float`, `Vector2`, `List<T>`）构成，严格适配 Unity 原生 `JsonUtility`，杜绝第三方重量级二进制或反射序列化库，零 GC 侵入。
3. **确定性着陆（Deterministic Resolution）**：
   在途瞬态实体（抛物线空中的水球/土球）直接丢弃；正在飞行未落地的掉落物（`WaterBallDrop`）强制瞬移至终点落点 `Landing` 并就地置为 `Waiting` 静态物理态保存。
4. **故障安全写盘（Atomic File Replacement）**：
   磁盘写入强制遵循 `save.json.tmp` $\rightarrow$ `File.Replace` 流程，杜绝游戏意外退出或掉电导致的坏档、空档。

## 4. URP 2D 后处理捕获管线 (`SaveCaptureFeature`)

### 4.1 捕获机制与后处理保真
- **执行时机**：`RenderPassEvent.AfterRenderingPostProcessing`。
  此时 URP 2D 管线已完成全部 2D 光照、Bloom 泛光、暗角及颜色校正计算，且**尚未开始渲染 Canvas UGUI**。
- **天然优势**：截图 100% 还原游戏真实画面美感，且**绝对不会包含任何打开的暂停菜单、按钮等 UI 元素**。

```text
URP 2D 渲染管线帧末
┌────────────────────────────────────────────────────────┐
│ 2D Light + Sprite 混合渲染                              │
├────────────────────────────────────────────────────────┤
│ URP Post-Processing 处理 (Bloom, Color, Vignette)      │
├────────────────────────────────────────────────────────┤
│ ★【SaveCapturePass 插入点】                            │
│  Blit(CameraColorTarget -> 320x180 离屏 RT)             │
│  发起 AsyncGPUReadback (GPU 零停顿回读)                │
├────────────────────────────────────────────────────────┤
│ 渲染 UGUI Canvas (PausePanel / SavePanel) - [不被截入]  │
└────────────────────────────────────────────────────────┘
```

### 4.2 工业级抓取 Pass 实现
```csharp
using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Rendering.Universal;

namespace DeepseaOil.Presentation.RenderPass
{
    public sealed class SaveCaptureFeature : ScriptableRendererFeature
    {
        private sealed class SaveCapturePass : ScriptableRenderPass
        {
            private static readonly int ThumbWidth = 320;
            private static readonly int ThumbHeight = 180;

            private RenderTexture _thumbRT;
            private System.Action<byte[]> _onPngReady;
            private bool _captureRequested;

            public SaveCapturePass()
            {
                renderPassEvent = RenderPassEvent.AfterRenderingPostProcessing;
            }

            public void RequestCapture(System.Action<byte[]> onPngReady)
            {
                _onPngReady = onPngReady;
                _captureRequested = true;
            }

            public override void Execute(ScriptableRenderContext context, ref RenderingData renderingData)
            {
                if (!_captureRequested || renderingData.cameraData.cameraType != CameraType.Game) return;
                _captureRequested = false;

                if (_thumbRT == null)
                {
                    _thumbRT = new RenderTexture(ThumbWidth, ThumbHeight, 0, RenderTextureFormat.ARGB32)
                    {
                        name = "SaveThumb_RT",
                        filterMode = FilterMode.Bilinear
                    };
                }

                CommandBuffer cmd = CommandBufferPool.Get("SaveCapture");
                RTHandle source = renderingData.cameraData.renderer.cameraColorTargetHandle;

                // 硬件降采样 Blit
                Blit(cmd, source, _thumbRT);
                context.ExecuteCommandBuffer(cmd);
                CommandBufferPool.Release(cmd);

                // 异步 GPU 回读：避免主线程 ReadPixels 产生 20ms 尖峰卡顿
                AsyncGPUReadback.Request(_thumbRT, 0, TextureFormat.RGB24, OnReadbackCompleted);
            }

            private void OnReadbackCompleted(AsyncGPUReadbackRequest request)
            {
                if (request.hasError) return;

                // 在后台线程并发进行 PNG 编码，主线程完全脱离阻塞
                var data = request.GetData<byte>().ToArray();
                System.Threading.Tasks.Task.Run(() =>
                {
                    Texture2D tex = new Texture2D(ThumbWidth, ThumbHeight, TextureFormat.RGB24, false);
                    tex.LoadRawTextureData(data);
                    tex.Apply();
                    byte[] png = tex.EncodeToPNG();
                    Object.Destroy(tex);
                    _onPngReady?.Invoke(png);
                });
            }

            public void Dispose()
            {
                if (_thumbRT != null)
                {
                    _thumbRT.Release();
                    Object.Destroy(_thumbRT);
                    _thumbRT = null;
                }
            }
        }

        private static SaveCaptureFeature _instance;
        private SaveCapturePass _pass;

        public static void Capture(System.Action<byte[]> onPngReady)
        {
            _instance?._pass.RequestCapture(onPngReady);
        }

        public override void Create()
        {
            _instance = this;
            _pass = new SaveCapturePass();
        }

        public override void AddRenderPasses(ScriptableRenderer renderer, ref RenderingData renderingData)
        {
            renderer.EnqueuePass(_pass);
        }

        protected override void Dispose(bool disposing)
        {
            _pass?.Dispose();
            _instance = null;
        }
    }
}
```

### 5.3 活体敌人现场回填 (`CombatDirector.cs`)
```csharp
public List<EnemySaveData> ExportEnemies()
{
    var list = new List<EnemySaveData>(_enemies.Count);
    for (int i = 0; i < _enemies.Count; i++)
    {
        EnemyActor actor = _enemies[i];
        if (actor == null || !actor.IsAlive) continue;

        list.Add(new EnemySaveData
        {
            EnemyId = actor.EnemyId,
            CurrentHp = actor.Hp,
            Position = actor.Position,
            Facing = actor.EngineVelocity.sqrMagnitude > 0f ? actor.EngineVelocity.normalized : Vector2.right
        });
    }
    return list;
}

public void RestoreEnemies(List<EnemySaveData> list)
{
    ClearAll(); // 彻底清理场景现有实体，不留幽灵怪
    if (list == null) return;

    for (int i = 0; i < list.Count; i++)
    {
        EnemySaveData data = list[i];
        EnemySpec spec = _specOf(data.EnemyId);
        if (spec == null) continue;

        GameObject prefab = AssetModule.Load<GameObject>($"enemies/Enemy_{data.EnemyId}");
        GameObject go = Object.Instantiate(prefab, data.Position, Quaternion.identity, _actorRoot);
        EnemyActor actor = go.GetComponent<EnemyActor>();

        // 挂接委托与组合根收养
        if (_onKilled != null) actor.Died = OnEnemyDied;
        _combatRoot.AdoptWaveEnemy(actor, data.Position, in spec, data.Facing);

        // 差值扣血，精准恢复残血状态
        int diffDamage = spec.Hp - data.CurrentHp;
        if (diffDamage > 0)
        {
            actor.Stats.ApplyDamage(diffDamage);
        }

        _enemies.Add(actor);
    }

    AliveCount = _enemies.Count;
    Announce();
}
```

public List<TileSaveData> ExportTiles()
{
    var list = new List<TileSaveData>(_machines.Count);
    foreach (var kv in _machines)
    {
        if (kv.Value.CurrentId == cfg.dso.TileStateType.Normal) continue;

        float remaining = -1f;
        if (kv.Value.Current is TableTileState state)
        {
            // 通过接口或属性暴露当前地块剩余秒数
            remaining = state.RemainingDuration;
        }

        list.Add(new TileSaveData
        {
            X = kv.Key.x,
            Y = kv.Key.y,
            StateId = (int)kv.Value.CurrentId,
            RemainingDuration = remaining
        });
    }
    return list;
}

public void RestoreTiles(List<TileSaveData> list)
{
    if (list == null) return;

    for (int i = 0; i < list.Count; i++)
    {
        TileSaveData data = list[i];
        Vector3Int cell = new Vector3Int(data.X, data.Y, 0);
        if (!_cells.Contains(cell)) continue;

        SwitchTileState(cell, (cfg.dso.TileStateType)data.StateId, data.RemainingDuration);
    }
}

// 故障安全原子落盘实现片段
private static void AtomicWriteFile(string path, string content)
{
    string tempPath = path + ".tmp";
    System.IO.File.WriteAllText(tempPath, content);
    if (System.IO.File.Exists(path))
    {
        System.IO.File.Replace(tempPath, path, null);
    }
    else
    {
        System.IO.File.Move(tempPath, path);
    }
}
