### 二、确定性修订方案与完整源码交付

#### 核心修订 1：解决角色“受光与闪白死锁” —— 终极 `CelLit.shader`
- **物理路径**：`Assets/Resources/materials/Shaders/CelLit.shader`
- **修订根因**：既有角色依赖 `ActorTintDriver` 向材质属性块写入 `_FlashAmount` 与 `_FlashColor`。必须将**纯白剪影插值逻辑无缝融合至 `CelLit.shader` 的末端**！
- **修复效果**：地表与角色可共用此 Shader；未挨打时角色接受赛璐珞环境光，挨打瞬间 0.12 秒强制插值覆盖为纯白剪影，闪白结束平滑淡回环境受光色，两套系统完全兼容。同时**彻底修正顶点着色器中的 `ComputeScreenPos` 传值算法**。

```hlsl
Shader "DeepseaOil/2D/CelLit"
{
    Properties
    {
        [PerRendererData] _MainTex ("Sprite Texture", 2D) = "white" {}
        _Color ("Tint", Color) = (1, 1, 1, 1)

        [Header(Cel Shading and Floor)]
        _AmbientFloor ("Ambient Floor", Range(0.35, 0.6)) = 0.35
        _CelBands ("Cel Bands Count", Range(2, 4)) = 3

        [Header(Rim and Metal Highlight)]
        [Toggle] _EnableRim ("Enable Top Rim Highlight", Float) = 0
        _RimThreshold ("Rim Threshold UV Y", Range(0.8, 1.0)) = 0.92
        _RimColor ("Rim Highlight Color", Color) = (0.976, 0.808, 0.353, 1) // #F9CE5A 明黄
        _MetalThreshold ("Metal Highlight Threshold", Range(0.7, 0.95)) = 0.85
        _MetalIntensity ("Metal Highlight Intensity", Range(0.0, 1.0)) = 0.0

        [Header(Hit Flash Integration)]
        _FlashAmount ("Flash Amount", Range(0.0, 1.0)) = 0.0
        _FlashColor ("Flash Color", Color) = (1, 1, 1, 1)
    }

    SubShader
    {
        Tags
        {
            "Queue" = "Transparent"
            "RenderType" = "Transparent"
            "RenderPipeline" = "UniversalPipeline"
            "IgnoreProjector" = "True"
        }

        Blend SrcAlpha OneMinusSrcAlpha
        Cull Off
        ZWrite Off

        Pass
        {
            Name "Universal2D"
            Tags { "LightMode" = "Universal2D" }

            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #pragma multi_compile USE_SHAPE_LIGHT_TYPE_0 __

            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            struct Attributes
            {
                float3 positionOS : POSITION;
                float4 color : COLOR;
                float2 uv : TEXCOORD0;
            };

            struct Varyings
            {
                float4 positionCS : SV_POSITION;
                float4 color : COLOR;
                float2 uv : TEXCOORD0;
                float4 screenUV : TEXCOORD1;
            };

            TEXTURE2D(_MainTex);
            SAMPLER(sampler_MainTex);

            #if USE_SHAPE_LIGHT_TYPE_0
            TEXTURE2D(_ShapeLightTexture0);
            SAMPLER(sampler_ShapeLightTexture0);
            #endif

            CBUFFER_START(UnityPerMaterial)
                float4 _Color;
                float4 _MainTex_ST;
                float4 _RimColor;
                float4 _FlashColor;
                float _FlashAmount;
                float _AmbientFloor;
                float _CelBands;
                float _EnableRim;
                float _RimThreshold;
                float _MetalThreshold;
                float _MetalIntensity;
            CBUFFER_END

            Varyings vert(Attributes input)
            {
                Varyings output = (Varyings)0;
                output.positionCS = TransformObjectToHClip(input.positionOS);
                output.uv = TRANSFORM_TEX(input.uv, _MainTex);
                output.color = input.color * _Color;

                // 官方正交/透视标准屏幕坐标输出：严禁在顶点阶段自行除以 w
                output.screenUV = ComputeScreenPos(output.positionCS);
                return output;
            }

            half4 frag(Varyings input) : SV_Target
            {
                half4 mainTex = SAMPLE_TEXTURE2D(_MainTex, sampler_MainTex, input.uv);
                half4 baseCol = mainTex * input.color;

                #if USE_SHAPE_LIGHT_TYPE_0
                // 片元透视除法获取准确归一化屏幕 UV
                float2 lightUV = input.screenUV.xy / input.screenUV.w;
                half4 light = SAMPLE_TEXTURE2D(_ShapeLightTexture0, sampler_ShapeLightTexture0, lightUV);
                #else
                half4 light = half4(1, 1, 1, 1);
                #endif

                // 1. 暗部下限锁定：防止弱光跌入死黑
                light.rgb = max(light.rgb, half3(_AmbientFloor, _AmbientFloor, _AmbientFloor));

                // 2. 赛璐珞光强阶梯量化
                half lightLum = dot(light.rgb, half3(0.299, 0.587, 0.114));
                half steppedLum = floor(lightLum * _CelBands) / _CelBands + (0.5 / _CelBands);
                half3 celLight = light.rgb * (steppedLum / max(lightLum, 0.001));

                half3 finalRgb = baseCol.rgb * celLight;

                // 3. 顶端 UV 伪高光亮线
                if (_EnableRim > 0.5 && input.uv.y > _RimThreshold)
                {
                    finalRgb = lerp(finalRgb, _RimColor.rgb, 0.65 * baseCol.a);
                }

                // 4. 金属亮度阈值阶梯反光
                half pixLum = dot(baseCol.rgb, half3(0.299, 0.587, 0.114));
                half metalMask = step(_MetalThreshold, pixLum) * _MetalIntensity;
                finalRgb = lerp(finalRgb, half3(1, 1, 1), metalMask);

                // 5. 受击纯白剪影材质插值（最高优先级）：不受环境光压暗影响
                finalRgb = lerp(finalRgb, _FlashColor.rgb, _FlashAmount);

                finalRgb *= baseCol.a;
                return half4(finalRgb, baseCol.a);
            }
            ENDHLSL
        }
    }
}
```

#### 核心修订 2：修复光照 UV 采样公式 —— 终极 `TileFlow.shader`
- **物理路径**：`Assets/Resources/materials/Shaders/TileFlow.shader`
- **修订根因**：修正原版本中在顶点着色器提前透视除法导致的屏幕光照采样拉伸错误。

```hlsl
Shader "DeepseaOil/2D/TileFlow"
{
    Properties
    {
        [PerRendererData] _MainTex ("Sprite Texture", 2D) = "white" { }
        _Color ("Tint", Color) = (1, 1, 1, 1)
        _FlowSpeed ("Flow Speed", Float) = 3
        _FlowAmplitude ("Flow Amplitude", Range(0, 0.05)) = 0.012
        _FlowFrequency ("Flow Frequency", Float) = 20
        _PhaseNoise ("Phase Noise", Range(0, 1)) = 0
        _AmbientFloor ("Ambient Floor", Range(0.35, 0.6)) = 0.35
    }

    SubShader
    {
        Tags
        {
            "RenderType" = "Transparent"
            "Queue" = "Transparent"
            "RenderPipeline" = "UniversalPipeline"
            "IgnoreProjector" = "True"
        }

        Blend SrcAlpha OneMinusSrcAlpha
        ZWrite Off
        Cull Off

        Pass
        {
            Name "Universal2D"
            Tags { "LightMode" = "Universal2D" }

            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #pragma multi_compile USE_SHAPE_LIGHT_TYPE_0 __

            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            struct Attributes
            {
                float4 positionOS : POSITION;
                float4 color : COLOR;
                float2 uv : TEXCOORD0;
            };

            struct Varyings
            {
                float4 positionCS : SV_POSITION;
                float4 color : COLOR;
                float2 uv : TEXCOORD0;
                float2 worldPos : TEXCOORD1;
                float4 screenUV : TEXCOORD2;
            };

            TEXTURE2D(_MainTex);
            SAMPLER(sampler_MainTex);

            #if USE_SHAPE_LIGHT_TYPE_0
            TEXTURE2D(_ShapeLightTexture0);
            SAMPLER(sampler_ShapeLightTexture0);
            #endif

            CBUFFER_START(UnityPerMaterial)
                float4 _Color;
                float4 _MainTex_ST;
                float _FlowSpeed;
                float _FlowAmplitude;
                float _FlowFrequency;
                float _PhaseNoise;
                float _AmbientFloor;
            CBUFFER_END

            Varyings vert(Attributes input)
            {
                Varyings output = (Varyings)0;
                output.positionCS = TransformObjectToHClip(input.positionOS.xyz);
                output.color = input.color * _Color;
                output.uv = TRANSFORM_TEX(input.uv, _MainTex);
                output.worldPos = TransformObjectToWorld(input.positionOS.xyz).xy;

                // 标准齐次裁剪坐标转换
                output.screenUV = ComputeScreenPos(output.positionCS);
                return output;
            }

            half4 frag(Varyings input) : SV_Target
            {
                float2 uv = input.uv;

                float2 cellPos = floor(input.worldPos);
                float hash = frac(sin(dot(cellPos, float2(12.9898, 78.233))) * 43758.5453);
                float phase = hash * _PhaseNoise * 6.2831853;

                float dx = sin(_Time.y * _FlowSpeed + uv.y * _FlowFrequency + phase) * _FlowAmplitude;
                float dy = sin(_Time.y * _FlowSpeed * 1.3 + uv.x * _FlowFrequency * 0.7 + phase) * _FlowAmplitude;

                half4 baseCol = SAMPLE_TEXTURE2D(_MainTex, sampler_MainTex, uv);
                float lum = dot(baseCol.rgb, float3(0.299, 0.587, 0.114));
                float darkWeight = saturate(1.0 - lum);

                float2 flowUV = uv + float2(dx, dy) * darkWeight;
                half4 flowedColor = SAMPLE_TEXTURE2D(_MainTex, sampler_MainTex, flowUV) * input.color;

                #if USE_SHAPE_LIGHT_TYPE_0
                float2 lightUV = input.screenUV.xy / input.screenUV.w;
                half4 light = SAMPLE_TEXTURE2D(_ShapeLightTexture0, sampler_ShapeLightTexture0, lightUV);
                #else
                half4 light = half4(1, 1, 1, 1);
                #endif

                light.rgb = max(light.rgb, half3(_AmbientFloor, _AmbientFloor, _AmbientFloor));

                half3 finalRgb = flowedColor.rgb * light.rgb;
                finalRgb *= flowedColor.a;

                return half4(finalRgb, flowedColor.a);
            }
            ENDHLSL
        }
    }
}
```

#### 核心修订 3：立面覆盖物接入 `Actor` 受光图层 —— `TileOverlayDirector.cs`
- **物理路径**：`Assets/Scripts/Presentation/Grid/TileOverlayDirector.cs`
- **修订根因**：原代码硬编码 `sr.sortingLayerID = RenderOrder.OverlayLayer;`（值等于 0 即 Default 层），导致立面物体完全接收不到 Light2D 的照射。
- **改动位置**：`TileOverlayDirector.CreateOverlayObject` 方法。

```csharp
        private GameObject CreateOverlayObject()
        {
            var go = new GameObject("TileOverlay");
            go.transform.SetParent(transform, false);

            var sr = go.AddComponent<SpriteRenderer>();
            
            // 修复：接入 Actor 物理受光层，严禁滞留在 Default 层导致不受光
            sr.sortingLayerName = DeepseaOil.Presentation.Visual.SortingLayerConsts.Actor;

            go.SetActive(false);
            return go;
        }
```

#### 核心修订 4：点光源生命周期收口防泄漏 —— `CombatRoot.cs` 接线
- **物理路径**：`Assets/Scripts/Presentation/CombatRoot.cs`
- **修订根因**：`CombatRoot.ClearAll()` 在重开与玩家复活时被调用，但漏掉了对 `TileLightDirector` 的清理，导致点光源常驻泄漏。
- **改动位置**：增加字段引用并在 `ClearAll()` 中收口。

```csharp
        // 1. 在可选接线区域新增点光源管理器引用（留空运行时自愈）
        [SerializeField] private DeepseaOil.Presentation.Lighting.TileLightDirector tileLightDirector = default;

        // 2. 在 ClearAll() 方法中补全收口调用
        public void ClearAll()
        {
            if (_combat != null) _combat.ClearAll();
            if (_levelLogic != null)
            {
                _levelLogic.Reset();
                _levelLogic.Start();
            }

            // 修复：必须清理动态点光源，防止复活后上一局的火/电光源常驻死锁池位
            if (tileLightDirector != null)
            {
                tileLightDirector.ClearAll();
            }
            else
            {
                var director = FindObjectOfType<DeepseaOil.Presentation.Lighting.TileLightDirector>();
                director?.ClearAll();
            }

            _balls?.ClearAll();
            _drops?.ClearAll();
            TileChainReactor.Clear();
            gridView?.ClearAll();

            // ... 其余既有敌人清理逻辑保持不变 ...
        }
```
