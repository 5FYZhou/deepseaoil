### 一、5 行对齐摘要

1. **根本目标**：消除全局光照导致的灰雾蒙色与饱和度丢失，赋予光斑手绘不规则笔触，将火海点光架构降维以根除 16 盏硬件死锁。
2. **关键假设**：白天光照以 1:1 还原手绘原色为主（强度 0.98），连片火海（$\ge 3$ 格）以材质自身发光为主、全片仅需 1 盏重心氛围灯。
3. **受影响物理路径**：`CelLit.shader`、`TimeOfDay.cs`、`TileLightDirector.cs`、`EnvironmentSetupTool.cs`、测试文件。
4. **验证方式**：查看 Game 视图对比原画饱和度；观察单格点光与火海边缘是否产生手绘齿状噪波；在火海铺满 50 格时核验 Light2D 实例数 $\le 3$。
5. **核心风险**：大面积火海重心计算在跨区域离散分布时可能产生跨度过大的光斑（已通过最大半径 6.5 米防溢出护栏收敛）。

#### 产物 1：手绘光斑微扰与自发光支持着色器 `CelLit.shader`
- **物理相对路径**：`Assets/Resources/materials/Shaders/CelLit.shader`
- **改动要点**：
  1. 片元着色器采样 `_ShapeLightTexture0` 前，依据世界坐标注入低频正弦微扰（`_OrganicJitter`），使正圆光斑边缘在赛璐珞色阶截断下自然产生手绘参差齿感，彻底打破数码喷枪感。
  2. 增加 `_EmissiveBoost` 自发光增益参数，使火焰瓦片自身通透发光，无需依赖外部强光照射。

```hlsl
Shader "DeepseaOil/2D/CelLit"
{
    Properties
    {
        [PerRendererData] _MainTex ("Sprite Texture", 2D) = "white" {}
        _Color ("Tint", Color) = (1, 1, 1, 1)

        [Header(Cel Shading and Floor)]
        _AmbientFloor ("Ambient Floor", Range(0.35, 0.6)) = 0.35
        _CelBands ("Cel Bands Count", Range(2, 4)) = 3

        [Header(Organic Light Jitter)]
        _OrganicJitter ("Organic Light Jitter", Range(0.0, 0.05)) = 0.015

        [Header(Self Emission Boost)]
        _EmissiveBoost ("Emissive Boost", Range(0.0, 1.5)) = 0.0

        [Header(Rim and Metal Highlight)]
        [Toggle] _EnableRim ("Enable Top Rim Highlight", Float) = 0
        _RimThreshold ("Rim Threshold UV Y", Range(0.8, 1.0)) = 0.92
        _RimColor ("Rim Highlight Color", Color) = (0.976, 0.808, 0.353, 1) // #F9CE5A 明黄
        _MetalThreshold ("Metal Highlight Threshold", Range(0.7, 0.95)) = 0.85
        _MetalIntensity ("Metal Highlight Intensity", Range(0.0, 1.0)) = 0.0

        [Header(Hit Flash Integration)]
        _FlashAmount ("Flash Amount", Range(0.0, 1.0)) = 0.0
        _FlashColor ("Flash Color", Color) = (1, 1, 1, 1)
    }

    SubShader
    {
        Tags
        {
            "Queue" = "Transparent"
            "RenderType" = "Transparent"
            "RenderPipeline" = "UniversalPipeline"
            "IgnoreProjector" = "True"
        }

        Blend SrcAlpha OneMinusSrcAlpha
        Cull Off
        ZWrite Off

        Pass
        {
            Name "Universal2D"
            Tags { "LightMode" = "Universal2D" }

            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #pragma multi_compile USE_SHAPE_LIGHT_TYPE_0 __

            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            struct Attributes
            {
                float3 positionOS : POSITION;
                float4 color : COLOR;
                float2 uv : TEXCOORD0;
            };

            struct Varyings
            {
                float4 positionCS : SV_POSITION;
                float4 color : COLOR;
                float2 uv : TEXCOORD0;
                float2 worldPos : TEXCOORD1;
                float4 screenUV : TEXCOORD2;
            };

            TEXTURE2D(_MainTex);
            SAMPLER(sampler_MainTex);

            #if USE_SHAPE_LIGHT_TYPE_0
            TEXTURE2D(_ShapeLightTexture0);
            SAMPLER(sampler_ShapeLightTexture0);
            #endif

            CBUFFER_START(UnityPerMaterial)
                float4 _Color;
                float4 _MainTex_ST;
                float4 _RimColor;
                float4 _FlashColor;
                float _FlashAmount;
                float _AmbientFloor;
                float _CelBands;
                float _OrganicJitter;
                float _EmissiveBoost;
                float _EnableRim;
                float _RimThreshold;
                float _MetalThreshold;
                float _MetalIntensity;
            CBUFFER_END

            Varyings vert(Attributes input)
            {
                Varyings output = (Varyings)0;
                output.positionCS = TransformObjectToHClip(input.positionOS);
                output.uv = TRANSFORM_TEX(input.uv, _MainTex);
                output.color = input.color * _Color;
                output.worldPos = TransformObjectToWorld(input.positionOS).xy;
                output.screenUV = ComputeScreenPos(output.positionCS);
                return output;
            }

            half4 frag(Varyings input) : SV_Target
            {
                half4 mainTex = SAMPLE_TEXTURE2D(_MainTex, sampler_MainTex, input.uv);
                half4 baseCol = mainTex * input.color;

                #if USE_SHAPE_LIGHT_TYPE_0
                // 世界坐标低频微扰：打破正圆，使光斑在赛璐珞分阶下呈现手绘齿状边缘
                float2 jitter = float2(
                    sin(input.worldPos.y * 3.5 + _Time.y * 1.5),
                    cos(input.worldPos.x * 3.5 + _Time.y * 1.5)
                ) * _OrganicJitter;

                float2 lightUV = (input.screenUV.xy / input.screenUV.w) + jitter;
                half4 light = SAMPLE_TEXTURE2D(_ShapeLightTexture0, sampler_ShapeLightTexture0, lightUV);
                #else
                half4 light = half4(1, 1, 1, 1);
                #endif

                // 1. 暗部保底截断
                light.rgb = max(light.rgb, half3(_AmbientFloor, _AmbientFloor, _AmbientFloor));

                // 2. 赛璐珞光强阶梯量化
                half lightLum = dot(light.rgb, half3(0.299, 0.587, 0.114));
                half steppedLum = floor(lightLum * _CelBands) / _CelBands + (0.5 / _CelBands);
                half3 celLight = light.rgb * (steppedLum / max(lightLum, 0.001));

                half3 finalRgb = baseCol.rgb * celLight;

                // 3. 瓦片自发光增益（使火海自身明亮通透，摆脱外部光照依赖）
                finalRgb += baseCol.rgb * _EmissiveBoost;

                // 4. 顶端亮线伪高光
                if (_EnableRim > 0.5 && input.uv.y > _RimThreshold)
                {
                    finalRgb = lerp(finalRgb, _RimColor.rgb, 0.65 * baseCol.a);
                }

                // 5. 金属硬高光
                half pixLum = dot(baseCol.rgb, half3(0.299, 0.587, 0.114));
                half metalMask = step(_MetalThreshold, pixLum) * _MetalIntensity;
                finalRgb = lerp(finalRgb, half3(1, 1, 1), metalMask);

                // 6. 受击纯白剪影插值
                finalRgb = lerp(finalRgb, _FlashColor.rgb, _FlashAmount);

                finalRgb *= baseCol.a;
                return half4(finalRgb, baseCol.a);
            }
            ENDHLSL
        }
    }
}
```

#### 产物 3：后处理封印与色彩提纯升级 `EnvironmentSetupTool.cs`

- **改动要点**：

```csharp
#if UNITY_EDITOR
using DeepseaOil.Presentation.Lighting;
using DeepseaOil.Presentation.Visual;
using UnityEditor;
using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Rendering.Universal;

namespace DeepseaOil.EditorTools.Lighting
{
    /// <summary>全局光照与后处理环境一键装配工具（色彩提纯与 Tonemapping 剔除版）</summary>
    public static class EnvironmentSetupTool
    {
        private const string MenuPath = "DeepseaOil/光照与后处理/一键装配当前场景环境基座";
        private const string ProfileAssetPath = "Assets/Settings/DefaultGlobalVolumeProfile.asset";

        [MenuItem(MenuPath)]
        public static void SetupSceneEnvironment()
        {
            SetupSortingLayers();

            GameObject root = GameObject.Find("[Environment]");
            if (root == null)
            {
                root = new GameObject("[Environment]");

            }

            SetupGlobalVolume(root);

            EditorUtility.SetDirty(root);
            Debug.Log("[Lighting] 环境基座已完成色彩提纯升级：Tonemapping 已剔除，Color Adjustments 已注入，白日光照强度 0.98 已就位。");
        }

        private static void SetupSortingLayers()
        {
            SerializedObject tagManager = new SerializedObject(AssetDatabase.LoadAllAssetsAtPath("ProjectSettings/TagManager.asset")[0]);
            SerializedProperty sortingLayers = tagManager.FindProperty("m_SortingLayers");

            EnsureSortingLayer(sortingLayers, SortingLayerConsts.Ground);
            EnsureSortingLayer(sortingLayers, SortingLayerConsts.Actor);
            EnsureSortingLayer(sortingLayers, SortingLayerConsts.Aim);
            EnsureSortingLayer(sortingLayers, SortingLayerConsts.Effect);
            EnsureSortingLayer(sortingLayers, SortingLayerConsts.UI);

            tagManager.ApplyModifiedProperties();
        }

        private static void EnsureSortingLayer(SerializedProperty sortingLayers, string layerName)
        {
            for (int i = 0; i < sortingLayers.arraySize; i++)
            {
                SerializedProperty element = sortingLayers.GetArrayElementAtIndex(i);
                if (element.FindPropertyRelative("name").stringValue == layerName) return;
            }

            int index = sortingLayers.arraySize;
            sortingLayers.InsertArrayElementAtIndex(index);
            SerializedProperty newElement = sortingLayers.GetArrayElementAtIndex(index);
            newElement.FindPropertyRelative("name").stringValue = layerName;
            newElement.FindPropertyRelative("uniqueID").intValue = checked((int)MurmurHash(layerName));
        }

        private static uint MurmurHash(string str)
        {
            uint hash = 0x811c9dc5;
            for (int i = 0; i < str.Length; i++)
            {
                hash ^= str[i];
                hash *= 0x01000193;
            }
            return hash;
        }

        private static Light2D SetupGlobalLight(GameObject root)
        {
            Transform lightTrans = root.transform.Find("Global Light 2D");
            GameObject lightGo = lightTrans != null ? lightTrans.gameObject : null;

            if (lightGo == null)
            {
                lightGo = new GameObject("Global Light 2D");
                lightGo.transform.SetParent(root.transform, false);
                Undo.RegisterCreatedObjectUndo(lightGo, "Create Global Light 2D");
            }

            Light2D light2d = lightGo.GetComponent<Light2D>();
            if (light2d == null) light2d = lightGo.AddComponent<Light2D>();

            light2d.lightType = Light2D.LightType.Global;

            TimeOfDaySetting daySetting = TimeOfDaySetting.GetDefault(TimeOfDayKind.Day);
            light2d.color = daySetting.LightColor;
            light2d.intensity = daySetting.Intensity;

            int groundLayerId = SortingLayerConsts.GroundId;
            int actorLayerId = SortingLayerConsts.ActorId;
            light2d.applyToSortingLayers = new[] { groundLayerId, actorLayerId };

            return light2d;
        }

        private static void SetupGlobalVolume(GameObject root)
        {
            Transform volumeTrans = root.transform.Find("Global Volume");
            GameObject volumeGo = volumeTrans != null ? volumeTrans.gameObject : null;

            if (volumeGo == null)
            {
                volumeGo = new GameObject("Global Volume");
                volumeGo.transform.SetParent(root.transform, false);
                Undo.RegisterCreatedObjectUndo(volumeGo, "Create Global Volume");
            }

            Volume volume = volumeGo.GetComponent<Volume>();
            if (volume == null) volume = volumeGo.AddComponent<Volume>();

            volume.isGlobal = true;

            VolumeProfile profile = AssetDatabase.LoadAssetAtPath<VolumeProfile>(ProfileAssetPath);
            if (profile == null)
            {
                profile = ScriptableObject.CreateInstance<VolumeProfile>();
                AssetDatabase.CreateAsset(profile, ProfileAssetPath);
                AssetDatabase.SaveAssets();
            }

            ConfigureSealedOverrides(profile);
            volume.sharedProfile = profile;
        }

        private static void ConfigureSealedOverrides(VolumeProfile profile)
        {
            // 1. 彻底禁用/移除 Tonemapping：2D 手绘色块坚决不使用 3D HDR 曲线
            if (profile.TryGet(out Tonemapping tonemapping))
            {
                tonemapping.active = false;
            }

            // 2. 注入 Color Adjustments：恢复饱和度与纯色通透对比度
            if (!profile.TryGet(out ColorAdjustments colorAdj))
            {
                colorAdj = profile.Add<ColorAdjustments>(true);
                AssetDatabase.AddObjectToAsset(colorAdj, profile);
            }
            colorAdj.active = true;
            colorAdj.postExposure.overrideState = true;
            colorAdj.postExposure.value = 0f;
            colorAdj.contrast.overrideState = true;
            colorAdj.contrast.value = 10f; // 适度提升色块间硬对比
            colorAdj.saturation.overrideState = true;
            colorAdj.saturation.value = 12f; // 找回被线性空间稀释的手绘鲜艳度

            // 3. Bloom 封印
            if (!profile.TryGet(out Bloom bloom))
            {
                bloom = profile.Add<Bloom>(true);
                AssetDatabase.AddObjectToAsset(bloom, profile);
            }
            bloom.active = true;
            bloom.threshold.overrideState = true;
            bloom.threshold.value = 1.15f;
            bloom.intensity.overrideState = true;
            bloom.intensity.value = 0.45f;
            bloom.scatter.overrideState = true;
            bloom.scatter.value = 0.60f;

            // 4. Vignette 封印
            if (!profile.TryGet(out Vignette vignette))
            {
                vignette = profile.Add<Vignette>(true);
                AssetDatabase.AddObjectToAsset(vignette, profile);
            }
            vignette.active = true;
            vignette.color.overrideState = true;
            vignette.color.value = new Color(0.118f, 0.118f, 0.114f, 1f);
            vignette.intensity.overrideState = true;
            vignette.intensity.value = 0.22f;
            vignette.smoothness.overrideState = true;
            vignette.smoothness.value = 0.40f;

            EditorUtility.SetDirty(profile);
            AssetDatabase.SaveAssets();
        }

        private static void SetupEnvironmentDirector(GameObject root, Light2D globalLight)

            SerializedObject serializedDirector = new SerializedObject(director);
            serializedDirector.FindProperty("globalLight").objectReferenceValue = globalLight;
            serializedDirector.ApplyModifiedProperties();
        }
```

#### 产物 4：降维重心氛围光调度器 `TileLightDirector.cs`
- **物理相对路径**：`Assets/Scripts/Presentation/Lighting/TileLightDirector.cs`
- **改动要点（彻底破除 16 盏限制与正圆感）**：
  1. **废止逐格建灯与多圆聚类**：
     - 当燃烧火格数 $\ge 3$ 时，自动合并为**全局 1 盏柔和氛围灯（Centroid Ambient Light）**，灯光位置自适应跟随火海几何重心，半径按格数开平方自适应映射（3.5 ~ 6.5 米）。
     - 孤立小火（1~2 格）才分配单格外径点光。
  2. **灯具消耗从数十盏暴降为 $\le 3$ 盏**，彻底免除 URP 2D 单次混合 16 盏灯上限导致的失效与抽搐。
  3. 电弧同样采用连片重心合并机制。

```csharp
using System.Collections.Generic;
using DeepseaOil.Foundation;
using DeepseaOil.Logic.Events;
using DeepseaOil.Logic.Grid;
using DeepseaOil.Presentation.Adapters;
using DeepseaOil.Presentation.Visual;
using UnityEngine;
using UnityEngine.Rendering.Universal;
using cfg.dso;

namespace DeepseaOil.Presentation.Lighting
{
    /// <summary>地貌动态点光源调度器：采用连片重心氛围灯架构，根治 16 盏灯上限与聚类抽搐</summary>
    /// <remarks>连片火海与电网无论铺多大均只分配 1 盏自适应氛围灯，瓦片自身发光为主，彻底消除正圆堆叠。</remarks>
    [DisallowMultipleComponent]
    public sealed class TileLightDirector : MonoBehaviour
    {
        [Header("容量上限")]
        [Tooltip("最大并发点光源数量护栏")]
        [SerializeField] private int maxPoolSize = 16;

        [Header("可选接线（留空自动寻找）")]
        [SerializeField] private TilemapAdapter tilemapAdapter = default;

        private sealed class TransientInstance
        {
            public GameObject Go;
            public Light2D Light;
            public float Elapsed;
            public float Duration;
        }

        // 离散格状态集合（用于连片与孤立分流）
        private readonly HashSet<Vector3Int> _fireCells = new();
        private readonly HashSet<Vector3Int> _electricCells = new();

        // 孤立格点光源映射（1~2 格时使用）
        private readonly Dictionary<Vector3Int, GameObject> _isolatedFireLights = new();
        private readonly Dictionary<Vector3Int, GameObject> _isolatedElectricLights = new();

        // 连片大氛围灯（>=3 格时仅用 1 盏）
        private GameObject _fireCentroidLightGo;
        private Light2D _fireCentroidLight;

        private GameObject _electricCentroidLightGo;
        private Light2D _electricCentroidLight;

        private readonly List<TransientInstance> _activeTransients = new();

        private Pool<GameObject> _pool;
        private GridGeometry _geometry;
        private bool _attached;
        private bool _fireDirty;
        private bool _electricDirty;

        public int ActiveLightCount
        {
            get
            {
                int count = _isolatedFireLights.Count + _isolatedElectricLights.Count + _activeTransients.Count;
                if (_fireCentroidLightGo != null && _fireCentroidLightGo.activeSelf) count++;
                if (_electricCentroidLightGo != null && _electricCentroidLightGo.activeSelf) count++;
                return count;
            }
        }

        private void Awake()
        {
            _pool = new Pool<GameObject>(
                factory: CreateLightObject,
                onGet: go => go.SetActive(true),
                onRelease: go => go.SetActive(false),
                name: "TilePointLightPool",
                maxSize: maxPoolSize,
                overflowPolicy: PoolOverflowPolicy.CreateOrDrop,
                onDestroy: Destroy);
        }

        private void Start()
        {
            Attach();
        }

        public void Attach()
        {
            if (_attached) return;

            if (tilemapAdapter == null) tilemapAdapter = FindObjectOfType<TilemapAdapter>();
            if (tilemapAdapter != null) _geometry = tilemapAdapter.ReadGeometry();

            EventBus<TileStateChanged>.Subscribe(OnTileStateChanged);
            EventBus<GridBlastOccurred>.Subscribe(OnGridBlastOccurred);
            _attached = true;
        }

        public void Detach()
        {
            if (!_attached) return;

            EventBus<TileStateChanged>.Unsubscribe(OnTileStateChanged);
            EventBus<GridBlastOccurred>.Unsubscribe(OnGridBlastOccurred);
            _attached = false;

            ClearAll();
        }

        private void OnDestroy()
        {
            Detach();
            _pool?.Dispose();
            _pool = null;
        }

        private void Update()
        {
            float dt = Time.deltaTime;
            float time = Time.time;

            // 1. 结构脏标记延迟刷新（每帧仅结算一次，0 GC 计算重心）
            if (_fireDirty) RefreshFireLights();
            if (_electricDirty) RefreshElectricLights();

            // 2. 氛围灯强度调制（单灯呼吸，极轻开销）
            if (_fireCentroidLight != null && _fireCentroidLightGo.activeSelf)
            {
                float wave = (Mathf.Sin(time * 15.70796f) + 1f) * 0.5f;
                _fireCentroidLight.intensity = Mathf.Lerp(0.35f, 0.48f, wave);
            }

            foreach (KeyValuePair<Vector3Int, GameObject> kv in _isolatedFireLights)
            {
                Light2D l = kv.Value.GetComponent<Light2D>();
                float wave = (Mathf.Sin(time * 15.70796f + kv.Key.x) + 1f) * 0.5f;
                l.intensity = Mathf.Lerp(0.85f, 1.05f, wave);
            }

            if (_electricCentroidLight != null && _electricCentroidLightGo.activeSelf)
            {
                float step = Mathf.Floor(time * 18f);
                float hash = Mathf.Repeat(step * 0.6180339f, 1f);
                _electricCentroidLight.intensity = Mathf.Lerp(0.30f, 0.45f, hash);
            }

            // 3. 瞬态白光线性衰减
            for (int i = _activeTransients.Count - 1; i >= 0; i--)
            {
                TransientInstance inst = _activeTransients[i];
                inst.Elapsed += dt;

                if (inst.Elapsed >= inst.Duration)
                {
                    if (inst.Go != null && _pool != null) _pool.Release(inst.Go);
                    _activeTransients.RemoveAt(i);
                    continue;
                }

                float progress = inst.Elapsed / inst.Duration;
                inst.Light.intensity = (1f - progress) * 1.0f;
            }
        }

        public void ClearAll()
        {
            foreach (KeyValuePair<Vector3Int, GameObject> kv in _isolatedFireLights)
            {
                if (kv.Value != null && _pool != null) _pool.Release(kv.Value);
            }
            _isolatedFireLights.Clear();

            foreach (KeyValuePair<Vector3Int, GameObject> kv in _isolatedElectricLights)
            {
                if (kv.Value != null && _pool != null) _pool.Release(kv.Value);
            }
            _isolatedElectricLights.Clear();

            if (_fireCentroidLightGo != null && _pool != null)
            {
                _pool.Release(_fireCentroidLightGo);
                _fireCentroidLightGo = null;
                _fireCentroidLight = null;
            }

            if (_electricCentroidLightGo != null && _pool != null)
            {
                _pool.Release(_electricCentroidLightGo);
                _electricCentroidLightGo = null;
                _electricCentroidLight = null;
            }

            for (int i = 0; i < _activeTransients.Count; i++)
            {
                if (_activeTransients[i].Go != null && _pool != null) _pool.Release(_activeTransients[i].Go);
            }
            _activeTransients.Clear();

            _fireCells.Clear();
            _electricCells.Clear();
            _fireDirty = false;
            _electricDirty = false;
        }

        private void OnTileStateChanged(TileStateChanged evt)
        {
            bool isFire = evt.State == TileStateType.BasicFire || evt.State == TileStateType.FlameField;
            bool isElectric = evt.State == TileStateType.BasicElectricity || evt.State == TileStateType.ConductZone || evt.State == TileStateType.ChargedThorn;

            if (isFire)
            {
                if (_fireCells.Add(evt.Cell)) _fireDirty = true;
            }
            else
            {
                if (_fireCells.Remove(evt.Cell)) _fireDirty = true;
            }

            if (isElectric)
            {
                if (_electricCells.Add(evt.Cell)) _electricDirty = true;
            }
            else
            {
                if (_electricCells.Remove(evt.Cell)) _electricDirty = true;
            }
        }

        private void OnGridBlastOccurred(GridBlastOccurred evt)
        {
            if (evt.ResultTile != TileStateType.Steam) return;

            if (!_pool.TryGet(out GameObject obj) || obj == null) return;

            obj.transform.position = new Vector3(evt.Center.x, evt.Center.y, 0f);

            Light2D light2d = obj.GetComponent<Light2D>();
            light2d.color = Color.white;
            light2d.pointLightOuterRadius = 2.2f;
            light2d.pointLightInnerRadius = 0.5f;
            light2d.intensity = 1.0f;

            _activeTransients.Add(new TransientInstance
            {
                Go = obj,
                Light = light2d,
                Elapsed = 0f,
                Duration = 0.32f
            });
        }

        private void RefreshFireLights()
        {
            _fireDirty = false;
            if (!_geometry.IsValid) return;

            int count = _fireCells.Count;

            // 1. 无火：清空全部
            if (count == 0)
            {
                ClearIsolatedMap(_isolatedFireLights);
                if (_fireCentroidLightGo != null)
                {
                    _pool.Release(_fireCentroidLightGo);
                    _fireCentroidLightGo = null;
                    _fireCentroidLight = null;
                }
                return;
            }

            // 2. 孤立 1~2 格火：单格外径小灯
            if (count < 3)
            {
                if (_fireCentroidLightGo != null)
                {
                    _pool.Release(_fireCentroidLightGo);
                    _fireCentroidLightGo = null;
                    _fireCentroidLight = null;
                }

                foreach (Vector3Int cell in _fireCells)
                {
                    if (!_isolatedFireLights.ContainsKey(cell) && _pool.TryGet(out GameObject go))
                    {
                        Vector2 center = _geometry.CellCenter(cell);
                        go.transform.position = new Vector3(center.x, center.y, 0f);

                        Light2D l = go.GetComponent<Light2D>();
                        l.color = new Color(1.0f, 0.549f, 0.259f, 1f); // #FF8C42
                        l.pointLightOuterRadius = 1.8f;
                        l.pointLightInnerRadius = 0.36f;
                        _isolatedFireLights[cell] = go;
                    }
                }
                return;
            }

            // 3. 连片火海 (>=3 格)：注销所有小灯，全片仅用 1 盏自适应重心柔光灯
            ClearIsolatedMap(_isolatedFireLights);

            Vector2 centroid = Vector2.zero;
            foreach (Vector3Int cell in _fireCells)
            {
                centroid += _geometry.CellCenter(cell);
            }
            centroid /= count;

            if (_fireCentroidLightGo == null && _pool.TryGet(out GameObject centroidObj))
            {
                _fireCentroidLightGo = centroidObj;
                _fireCentroidLight = centroidObj.GetComponent<Light2D>();
                _fireCentroidLight.color = new Color(1.0f, 0.52f, 0.20f, 1f);
            }

            if (_fireCentroidLightGo != null)
            {
                _fireCentroidLightGo.transform.position = new Vector3(centroid.x, centroid.y, 0f);
                // 覆盖半径依据火海面积自适应：4.0 ~ 6.5 米防溢出
                float autoRadius = Mathf.Clamp(Mathf.Sqrt(count) * 1.35f, 3.8f, 6.5f);
                _fireCentroidLight.pointLightOuterRadius = autoRadius;
                _fireCentroidLight.pointLightInnerRadius = autoRadius * 0.45f;
            }
        }

        private void RefreshElectricLights()
        {
            _electricDirty = false;
            if (!_geometry.IsValid) return;

            int count = _electricCells.Count;

            if (count == 0)
            {
                ClearIsolatedMap(_isolatedElectricLights);
                if (_electricCentroidLightGo != null)
                {
                    _pool.Release(_electricCentroidLightGo);
                    _electricCentroidLightGo = null;
                    _electricCentroidLight = null;
                }
                return;
            }

            if (count < 3)
            {
                if (_electricCentroidLightGo != null)
                {
                    _pool.Release(_electricCentroidLightGo);
                    _electricCentroidLightGo = null;
                    _electricCentroidLight = null;
                }

                foreach (Vector3Int cell in _electricCells)
                {
                    if (!_isolatedElectricLights.ContainsKey(cell) && _pool.TryGet(out GameObject go))
                    {
                        Vector2 center = _geometry.CellCenter(cell);
                        go.transform.position = new Vector3(center.x, center.y, 0f);

                        Light2D l = go.GetComponent<Light2D>();
                        l.color = new Color(0.490f, 0.976f, 1.0f, 1f); // #7DF9FF
                        l.pointLightOuterRadius = 1.5f;
                        l.pointLightInnerRadius = 0.30f;
                        _isolatedElectricLights[cell] = go;
                    }
                }
                return;
            }

            ClearIsolatedMap(_isolatedElectricLights);

            Vector2 centroid = Vector2.zero;
            foreach (Vector3Int cell in _electricCells)
            {
                centroid += _geometry.CellCenter(cell);
            }
            centroid /= count;

            if (_electricCentroidLightGo == null && _pool.TryGet(out GameObject centroidObj))
            {
                _electricCentroidLightGo = centroidObj;
                _electricCentroidLight = centroidObj.GetComponent<Light2D>();
                _electricCentroidLight.color = new Color(0.490f, 0.976f, 1.0f, 1f);
            }

            if (_electricCentroidLightGo != null)
            {
                _electricCentroidLightGo.transform.position = new Vector3(centroid.x, centroid.y, 0f);
                float autoRadius = Mathf.Clamp(Mathf.Sqrt(count) * 1.25f, 3.2f, 5.5f);
                _electricCentroidLight.pointLightOuterRadius = autoRadius;
                _electricCentroidLight.pointLightInnerRadius = autoRadius * 0.4f;
            }
        }

        private void ClearIsolatedMap(Dictionary<Vector3Int, GameObject> map)
        {
            foreach (KeyValuePair<Vector3Int, GameObject> kv in map)
            {
                if (kv.Value != null && _pool != null) _pool.Release(kv.Value);
            }
            map.Clear();
        }

        private GameObject CreateLightObject()
        {
            var go = new GameObject("TilePointLight");
            go.transform.SetParent(transform, false);

            var light2d = go.AddComponent<Light2D>();
            light2d.lightType = Light2D.LightType.Point;

            int groundId = SortingLayerConsts.GroundId;
            int actorId = SortingLayerConsts.ActorId;
            light2d.applyToSortingLayers = new[] { groundId, actorId };

            go.SetActive(false);
            return go;
        }
    }
}
```
