> **【模式】** 架构落地与工业级交付模式（Engineer & Architect Mode）  
> **【目标】** 彻底剔除 3D 点光源拟真逻辑，全量交付《泰拉瑞亚》网格光照场 +《饥荒》手绘撕裂高反差光照系统  
> **【核心假设】** 场景由 `TilemapAdapter` 提供网格几何；以微型 GPU 纹理传输光照；夜晚环境光设定为 0.20 极高反差  
> **【影响范围】** `Assets/Scripts/Presentation/Lighting/`、`Assets/Resources/materials/Shaders/`、`Assets/Editor/Lighting/`  
> **【风险与置信度】** 彻底摆脱 URP 2D 单次混合 16 盏灯限制；连片火海自适应融合 / 方案置信度：High

### 一、架构重构与网格光照流水线拓扑

本方案将光照的物理定义从“空间中的发光灯泡”降维为“**在瓦片网格中传播的光照场（Tile Grid Lightmap）**”，整体架构流水线如下：

```text
  【逻辑与地块事件】
  TileStateChanged (火池/火海/电源/导电区) ➔ 标记发光源网格
  GridLogic.IsObstacle (墙体/陶砖)        ➔ 标记光阻阻挡网格
                          │
                          ▼
  【TileGridLightmap (CPU 微秒级泛洪扩散)】
  1. 写入发光源核心色 (火: 暖橙红 2.5Hz 呼吸; 电: 青白 18Hz 闪烁)
  2. 4 邻域网格扩散迭代 3~4 步 (空气衰减 28%，遇到障碍物衰减 75%)
  3. 写入 64×32 微型光照纹理 (双线性过滤自动平滑网格走势)
                          │
                          ▼
  【GPU 全局纹理注入: _TileLightmap】
  - 设置全局网格世界参数: _TileLightOrigin, _TileLightSize
                          │
        ┌─────────────────┴─────────────────┐
        ▼                                   ▼
  【CelLit.shader (地面/立面/角色)】    【TileFlow.shader (流动水体)】
  - 采样 _TileLightmap                - 采样 _TileLightmap
  - 注入《饥荒》世界坐标正弦撕裂微扰  - 叠加水流 UV 扭曲
  - 与全局环境光 (白天 0.98/夜晚 0.20) 做高反差取大融合 (Max)
```

### 二、工业级完整源码交付

#### 产物 1：网格光照场调度器 `TileGridLightmap.cs`
- **物理相对路径**：`Assets/Scripts/Presentation/Lighting/TileGridLightmap.cs`
- **设计意图**：替代原先的 `TileLightDirector`。维护一张对齐地图的微型网格光照贴图，0 GC 执行曼哈顿衰减泛洪。连片火海天然融合为异形光区，**0 盏 Light2D 开销，永久废除 16 盏限制**。

```csharp
using System.Collections.Generic;
using DeepseaOil.Logic.Events;
using DeepseaOil.Logic.Grid;
using DeepseaOil.Presentation.Adapters;
using UnityEngine;
using cfg.dso;

namespace DeepseaOil.Presentation.Lighting
{
    /// <summary>仿泰拉瑞亚网格光照场：在瓦片网格中执行扩散与阻挡，输出全局光照贴图</summary>
    /// <remarks>彻底抛弃 URP 2D 点光源组件，连片火海自然融合，零灯具数量限制，无圆规塑料感。</remarks>
    [DisallowMultipleComponent]
    public sealed class TileGridLightmap : MonoBehaviour
    {
        private const int GridWidth = 64;
        private const int GridHeight = 36;
        private const int TotalCells = GridWidth * GridHeight;

        private static readonly int TileLightmapId = Shader.PropertyToID("_TileLightmap");
        private static readonly int TileLightmapParamsId = Shader.PropertyToID("_TileLightmapParams");

        [Header("接线")]
        [SerializeField] private TilemapAdapter tilemapAdapter = default;

        [Header("扩散配置")]
        [Tooltip("空气衰减率（每跨一格保留的光强比例）")]
        [Range(0.5f, 0.85f)]
        [SerializeField] private float airTransmission = 0.72f;

        [Tooltip("障碍物透过率（穿透墙体的光强比例）")]
        [Range(0.0f, 0.4f)]
        [SerializeField] private float obstacleTransmission = 0.20f;

        private readonly HashSet<Vector3Int> _fireCells = new();
        private readonly HashSet<Vector3Int> _electricCells = new();
        private readonly HashSet<Vector3Int> _obstacleCells = new();

        private readonly Color32[] _sourceColors = new Color32[TotalCells];
        private readonly Color32[] _diffusedColors = new Color32[TotalCells];
        private readonly byte[] _obstacleMap = new byte[TotalCells];

        private Texture2D _lightmapTexture;
        private GridGeometry _geometry;
        private Vector3Int _originOffset;
        private bool _attached;
        private bool _isDirty;

        private void Awake()
        {
            _lightmapTexture = new Texture2D(GridWidth, GridHeight, TextureFormat.RGBA32, false)
            {
                name = "TileGridLightmap_RT",
                filterMode = FilterMode.Bilinear,
                wrapMode = TextureWrapMode.Clamp
            };
        }

        private void Start()
        {
            Attach();
        }

        public void Attach()
        {
            if (_attached) return;

            if (tilemapAdapter == null) tilemapAdapter = FindObjectOfType<TilemapAdapter>();

            if (tilemapAdapter != null)
            {
                _geometry = tilemapAdapter.ReadGeometry();
                _originOffset = new Vector3Int(-GridWidth / 2, -GridHeight / 2, 0);
            }

            EventBus<TileStateChanged>.Subscribe(OnTileStateChanged);
            _attached = true;
            _isDirty = true;
        }

        public void Detach()
        {
            if (!_attached) return;

            EventBus<TileStateChanged>.Unsubscribe(OnTileStateChanged);
            _attached = false;
            ClearAll();
        }

        private void OnDestroy()
        {
            Detach();
            if (_lightmapTexture != null)
            {
                Destroy(_lightmapTexture);
                _lightmapTexture = null;
            }
        }

        private void Update()
        {
            float time = Time.time;

            // 无论是否有结构改动，火光呼吸与电光抽搐持续推进
            float firePulse = 0.88f + 0.12f * Mathf.Sin(time * 15.70796f);
            float elecStep = Mathf.Repeat(Mathf.Floor(time * 18f) * 0.6180339f, 1f);
            float elecPulse = 0.85f + 0.15f * elecStep;

            UpdateLightingBuffer(firePulse, elecPulse);
            PropagateLight();

            _lightmapTexture.SetPixels32(_diffusedColors);
            _lightmapTexture.Apply(false, false);

            PushShaderGlobals();
        }

        public void ClearAll()
        {
            _fireCells.Clear();
            _electricCells.Clear();
            _obstacleCells.Clear();

            for (int i = 0; i < TotalCells; i++)
            {
                _sourceColors[i] = new Color32(0, 0, 0, 0);
                _diffusedColors[i] = new Color32(0, 0, 0, 0);
                _obstacleMap[i] = 0;
            }

            if (_lightmapTexture != null)
            {
                _lightmapTexture.SetPixels32(_diffusedColors);
                _lightmapTexture.Apply(false, false);
            }
        }

        private void OnTileStateChanged(TileStateChanged evt)
        {
            bool isFire = evt.State == TileStateType.BasicFire || evt.State == TileStateType.FlameField;
            bool isElec = evt.State == TileStateType.BasicElectricity || evt.State == TileStateType.ConductZone || evt.State == TileStateType.ChargedThorn;
            bool isObstacle = evt.State == TileStateType.TerracottaBrick;

            if (isFire) _fireCells.Add(evt.Cell); else _fireCells.Remove(evt.Cell);
            if (isElec) _electricCells.Add(evt.Cell); else _electricCells.Remove(evt.Cell);
            if (isObstacle) _obstacleCells.Add(evt.Cell); else _obstacleCells.Remove(evt.Cell);

            _isDirty = true;
        }

        private void UpdateLightingBuffer(float firePulse, float elecPulse)
        {
            // 清理源缓冲
            for (int i = 0; i < TotalCells; i++)
            {
                _sourceColors[i] = new Color32(0, 0, 0, 0);
            }

            if (_isDirty)
            {
                for (int i = 0; i < TotalCells; i++) _obstacleMap[i] = 0;
                foreach (Vector3Int cell in _obstacleCells)
                {
                    int idx = CellToIndex(cell);
                    if (idx >= 0) _obstacleMap[idx] = 1;
                }
                _isDirty = false;
            }

            // 写入火光核心
            byte fireR = (byte)Mathf.Clamp(255 * firePulse, 0, 255);
            byte fireG = (byte)Mathf.Clamp(145 * firePulse, 0, 255);
            byte fireB = (byte)Mathf.Clamp(50 * firePulse, 0, 255);
            Color32 fireColor = new Color32(fireR, fireG, fireB, 255);

            foreach (Vector3Int cell in _fireCells)
            {
                int idx = CellToIndex(cell);
                if (idx >= 0) _sourceColors[idx] = fireColor;
            }

            // 写入电光核心
            byte elecR = (byte)Mathf.Clamp(125 * elecPulse, 0, 255);
            byte elecG = (byte)Mathf.Clamp(245 * elecPulse, 0, 255);
            byte elecB = (byte)Mathf.Clamp(255 * elecPulse, 0, 255);
            Color32 elecColor = new Color32(elecR, elecG, elecB, 255);

            foreach (Vector3Int cell in _electricCells)
            {
                int idx = CellToIndex(cell);
                if (idx >= 0) _sourceColors[idx] = elecColor;
            }
        }

        /// <summary>仿泰拉瑞亚 4 邻域双向扩散循环，支持墙体阴影阻挡</summary>
        private void PropagateLight()
        {
            // 拷贝基底
            for (int i = 0; i < TotalCells; i++)
            {
                _diffusedColors[i] = _sourceColors[i];
            }

            // 正向扫描（左上至右下）
            for (int y = 1; y < GridHeight - 1; y++)
            {
                int rowOffset = y * GridWidth;
                for (int x = 1; x < GridWidth - 1; x++)
                {
                    int i = rowOffset + x;
                    float trans = _obstacleMap[i] == 1 ? obstacleTransmission : airTransmission;

                    Color32 left = _diffusedColors[i - 1];
                    Color32 down = _diffusedColors[i - GridWidth];

                    byte maxR = (byte)(Mathf.Max(left.r, down.r) * trans);
                    byte maxG = (byte)(Mathf.Max(left.g, down.g) * trans);
                    byte maxB = (byte)(Mathf.Max(left.b, down.b) * trans);

                    if (maxR > _diffusedColors[i].r) _diffusedColors[i].r = maxR;
                    if (maxG > _diffusedColors[i].g) _diffusedColors[i].g = maxG;
                    if (maxB > _diffusedColors[i].b) _diffusedColors[i].b = maxB;
                }
            }

            // 反向扫描（右下至左上）
            for (int y = GridHeight - 2; y >= 1; y--)
            {
                int rowOffset = y * GridWidth;
                for (int x = GridWidth - 2; x >= 1; x--)
                {
                    int i = rowOffset + x;
                    float trans = _obstacleMap[i] == 1 ? obstacleTransmission : airTransmission;

                    Color32 right = _diffusedColors[i + 1];
                    Color32 up = _diffusedColors[i + GridWidth];

                    byte maxR = (byte)(Mathf.Max(right.r, up.r) * trans);
                    byte maxG = (byte)(Mathf.Max(right.g, up.g) * trans);
                    byte maxB = (byte)(Mathf.Max(right.b, up.b) * trans);

                    if (maxR > _diffusedColors[i].r) _diffusedColors[i].r = maxR;
                    if (maxG > _diffusedColors[i].g) _diffusedColors[i].g = maxG;
                    if (maxB > _diffusedColors[i].b) _diffusedColors[i].b = maxB;
                }
            }
        }

        private int CellToIndex(Vector3Int cell)
        {
            int x = cell.x - _originOffset.x;
            int y = cell.y - _originOffset.y;

            if (x < 0 || x >= GridWidth || y < 0 || y >= GridHeight) return -1;
            return y * GridWidth + x;
        }

        private void PushShaderGlobals()
        {
            Shader.SetGlobalTexture(TileLightmapId, _lightmapTexture);

            Vector2 originWorld = _geometry.CellCenter(_originOffset);
            float cellSize = _geometry.IsValid ? _geometry.CellSize : 1f;

            // X: OriginX, Y: OriginY, Z: TotalWorldWidth, W: TotalWorldHeight
            Shader.SetGlobalVector(TileLightmapParamsId, new Vector4(
                originWorld.x,
                originWorld.y,
                GridWidth * cellSize,
                GridHeight * cellSize));
        }
    }
}
```

#### 产物 2：网格光照与饥荒手绘撕裂着色器 `CelLit.shader`
- **物理相对路径**：`Assets/Resources/materials/Shaders/CelLit.shader`
- **改动要点**：
  1. **彻底拔除 URP 2D 点光源 `_ShapeLightTexture0` 依赖**，改用网格光照图 `_TileLightmap`。
  2. 融入《饥荒》手绘正弦边缘撕裂（`_OrganicJitter`）。
  3. **高反差黑夜融合**：夜间环境光压低至 0.20，火海直接以满亮爆发，暗黑与火光产生极致撕裂对比！

```hlsl
Shader "DeepseaOil/2D/CelLit"
{
    Properties
    {
        [PerRendererData] _MainTex ("Sprite Texture", 2D) = "white" {}
        _Color ("Tint", Color) = (1, 1, 1, 1)

        [Header(Cel Shading and Floor)]
        _AmbientFloor ("Ambient Floor", Range(0.2, 0.6)) = 0.20
        _CelBands ("Cel Bands Count", Range(2, 4)) = 3

        [Header(Organic Light Jitter)]
        _OrganicJitter ("Hand-drawn Edge Jitter", Range(0.0, 0.05)) = 0.018

        [Header(Rim and Metal Highlight)]
        [Toggle] _EnableRim ("Enable Top Rim Highlight", Float) = 0
        _RimThreshold ("Rim Threshold UV Y", Range(0.8, 1.0)) = 0.92
        _RimColor ("Rim Highlight Color", Color) = (0.976, 0.808, 0.353, 1)
        _MetalThreshold ("Metal Highlight Threshold", Range(0.7, 0.95)) = 0.85
        _MetalIntensity ("Metal Highlight Intensity", Range(0.0, 1.0)) = 0.0

        [Header(Hit Flash Integration)]
        _FlashAmount ("Flash Amount", Range(0.0, 1.0)) = 0.0
        _FlashColor ("Flash Color", Color) = (1, 1, 1, 1)
    }

    SubShader
    {
        Tags
        {
            "Queue" = "Transparent"
            "RenderType" = "Transparent"
            "RenderPipeline" = "UniversalPipeline"
            "IgnoreProjector" = "True"
        }

        Blend SrcAlpha OneMinusSrcAlpha
        Cull Off
        ZWrite Off

        Pass
        {
            Name "Universal2D"
            Tags { "LightMode" = "Universal2D" }

            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #pragma multi_compile USE_SHAPE_LIGHT_TYPE_0 __

            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            struct Attributes
            {
                float3 positionOS : POSITION;
                float4 color : COLOR;
                float2 uv : TEXCOORD0;
            };

            struct Varyings
            {
                float4 positionCS : SV_POSITION;
                float4 color : COLOR;
                float2 uv : TEXCOORD0;
                float2 worldPos : TEXCOORD1;
            };

            TEXTURE2D(_MainTex);
            SAMPLER(sampler_MainTex);

            TEXTURE2D(_TileLightmap);
            SAMPLER(sampler_TileLightmap);

            #if USE_SHAPE_LIGHT_TYPE_0
            TEXTURE2D(_ShapeLightTexture0);
            SAMPLER(sampler_ShapeLightTexture0);
            #endif

            CBUFFER_START(UnityPerMaterial)
                float4 _Color;
                float4 _MainTex_ST;
                float4 _RimColor;
                float4 _FlashColor;
                float _FlashAmount;
                float _AmbientFloor;
                float _CelBands;
                float _OrganicJitter;
                float _EnableRim;
                float _RimThreshold;
                float _MetalThreshold;
                float _MetalIntensity;
            CBUFFER_END

            float4 _TileLightmapParams; // X: OriginX, Y: OriginY, Z: Width, W: Height

            Varyings vert(Attributes input)
            {
                Varyings output = (Varyings)0;
                output.positionCS = TransformObjectToHClip(input.positionOS);
                output.uv = TRANSFORM_TEX(input.uv, _MainTex);
                output.color = input.color * _Color;
                output.worldPos = TransformObjectToWorld(input.positionOS).xy;
                return output;
            }

            half4 frag(Varyings input) : SV_Target
            {
                half4 mainTex = SAMPLE_TEXTURE2D(_MainTex, sampler_MainTex, input.uv);
                half4 baseCol = mainTex * input.color;

                // 1. 全局环境光基底（由 Global Light 2D 驱动昼夜，白天 0.98，夜晚 0.20）
                #if USE_SHAPE_LIGHT_TYPE_0
                float4 screenUV = ComputeScreenPos(input.positionCS);
                half4 envLight = SAMPLE_TEXTURE2D(_ShapeLightTexture0, sampler_ShapeLightTexture0, screenUV.xy / screenUV.w);
                #else
                half4 envLight = half4(1, 1, 1, 1);
                #endif
                envLight.rgb = max(envLight.rgb, half3(_AmbientFloor, _AmbientFloor, _AmbientFloor));

                // 2. 泰拉瑞亚网格光照场采样 + 饥荒手绘边缘抖动
                float2 gridUV = (input.worldPos - _TileLightmapParams.xy) / _TileLightmapParams.zw;

                // 世界坐标高频撕裂微扰：将平滑过渡转换为手绘纸质/锯齿边缘
                float2 handJitter = float2(
                    sin(input.worldPos.y * 5.0 + _Time.y * 1.5),
                    cos(input.worldPos.x * 5.0 + _Time.y * 1.5)
                ) * _OrganicJitter;

                half3 tileLight = SAMPLE_TEXTURE2D(_TileLightmap, sampler_TileLightmap, saturate(gridUV + handJitter)).rgb;

                // 3. 高反差合成：环境光为暗夜帷幕，网格光场为撕开的耀眼火光
                half3 mixedLight = max(envLight.rgb, tileLight * 1.25);

                // 4. 赛璐珞色阶离散化
                half lightLum = dot(mixedLight, half3(0.299, 0.587, 0.114));
                half steppedLum = floor(lightLum * _CelBands) / _CelBands + (0.5 / _CelBands);
                half3 finalLight = mixedLight * (steppedLum / max(lightLum, 0.001));

                half3 finalRgb = baseCol.rgb * finalLight;

                // 5. 顶端亮线伪高光
                if (_EnableRim > 0.5 && input.uv.y > _RimThreshold)
                {
                    finalRgb = lerp(finalRgb, _RimColor.rgb, 0.65 * baseCol.a);
                }

                // 6. 金属硬高光
                half pixLum = dot(baseCol.rgb, half3(0.299, 0.587, 0.114));
                half metalMask = step(_MetalThreshold, pixLum) * _MetalIntensity;
                finalRgb = lerp(finalRgb, half3(1, 1, 1), metalMask);

                // 7. 受击纯白剪影插值
                finalRgb = lerp(finalRgb, _FlashColor.rgb, _FlashAmount);

                finalRgb *= baseCol.a;
                return half4(finalRgb, baseCol.a);
            }
            ENDHLSL
        }
    }
}
```

#### 产物 3：高反差昼夜时段契约 `TimeOfDay.cs`

#### 产物 4：一键装配工具升级 `TileLightmapSetupTool.cs`
- **物理相对路径**：`Assets/Editor/Lighting/TileLightmapSetupTool.cs`
- **设计意图**：提供菜单命令，自动清除旧的 `TileLightDirector`（旧点光池），挂载全新的 `TileGridLightmap`，并完成与 `TilemapAdapter` 的自动绑定。

```csharp
#if UNITY_EDITOR
using DeepseaOil.Presentation.Adapters;
using DeepseaOil.Presentation.Lighting;
using UnityEditor;
using UnityEngine;

namespace DeepseaOil.EditorTools.Lighting
{
    public static class TileLightmapSetupTool
    {
        private const string MenuPath = "DeepseaOil/光照与后处理/切换至网格光照场系统(新范式)";

        [MenuItem(MenuPath)]
        public static void SetupTileGridLightmap()
        {
            GameObject env = GameObject.Find("[Environment]");
            if (env == null)
            {
                env = new GameObject("[Environment]");
                Undo.RegisterCreatedObjectUndo(env, "Create [Environment]");
            }

            // 彻底清除旧点光系统
            TileLightDirector oldDirector = env.GetComponent<TileLightDirector>();
            if (oldDirector != null)
            {
                Undo.DestroyObjectImmediate(oldDirector);
            }

            // 挂载全新的网格光照场
            TileGridLightmap lightmap = env.GetComponent<TileGridLightmap>();
            if (lightmap == null)
            {
                lightmap = env.AddComponent<TileGridLightmap>();
                Undo.RegisterCreatedObjectUndo(lightmap, "Add TileGridLightmap");
            }

            TilemapAdapter adapter = Object.FindObjectOfType<TilemapAdapter>();
            if (adapter != null)
            {
                SerializedObject so = new SerializedObject(lightmap);
                so.FindProperty("tilemapAdapter").objectReferenceValue = adapter;
                so.ApplyModifiedProperties();
            }

            EditorUtility.SetDirty(env);
            Debug.Log("[Lighting] 新范式装配完成：已卸载旧点光源池，TileGridLightmap 网格光照场已挂载并接线。");
        }
    }
}
#endif
```
