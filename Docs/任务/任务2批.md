### 一、材质受光算法与 Shader 分流架构

为满足《美术视觉工程化执行宪章》中**“赛璐珞色阶、暗部 $\ge 0.35$、边缘伪高光、禁止 PBR”**的硬指标，着色器数学模型确立如下：

```text
                        [片元输入: baseCol, lightingUV]
                                      │
            ┌─────────────────────────┴─────────────────────────┐
            ▼                                                   ▼
   【采样 2D 屏幕光照图】                               【片元纹理与属性提取】
   half4 light = _ShapeLightTexture0                     half lum = Dot(baseCol, 0.299, 0.587, 0.114)
            │                                                   │
            ▼                                                   ▼
   【暗部强制保底 (Ambient Floor)】                      【赛璐珞高光与边缘亮线】
   light.rgb = max(light.rgb, 0.35)                      metalHigh = step(0.85, lum)
            │                                            rimHigh = step(0.92, UV.y) * _RimColor
            ▼                                                   │
   【赛璐珞 3 阶量化 (Step-Banding)】                            │
   lightLum = Dot(light.rgb, LumaMask)                          │
   celScale = 阶梯离散化(lightLum)                               │
            │                                                   │
            └─────────────────────────┬─────────────────────────┘
                                      ▼
                      [最终输出: celLitColor + Highlights]
```

### 二、工业级完整源码交付

#### 产物 1：基础赛璐珞受光着色器 `CelLit.shader`
- **物理路径**：`Assets/Resources/materials/Shaders/CelLit.shader`
- **设计意图**：地表 Tilemap、立面 Overlay、角色通用的受光着色器。内置 0.35 暗部保底截断、3 阶赛璐珞光强量化及顶端 UV 伪高光。

```hlsl
Shader "DeepseaOil/2D/CelLit"
{
    Properties
    {
        [PerRendererData] _MainTex ("Sprite Texture", 2D) = "white" {}
        _Color ("Tint", Color) = (1, 1, 1, 1)

        [Header(Cel Shading and Floor)]
        _AmbientFloor ("Ambient Floor", Range(0.35, 0.6)) = 0.35
        _CelBands ("Cel Bands Count", Range(2, 4)) = 3

        [Header(Rim and Metal Highlight)]
        [Toggle] _EnableRim ("Enable Top Rim Highlight", Float) = 0
        _RimThreshold ("Rim Threshold UV Y", Range(0.8, 1.0)) = 0.92
        _RimColor ("Rim Highlight Color", Color) = (0.976, 0.808, 0.353, 1) // #F9CE5A 明黄
        _MetalThreshold ("Metal Highlight Threshold", Range(0.7, 0.95)) = 0.85
        _MetalIntensity ("Metal Highlight Intensity", Range(0.0, 1.0)) = 0.0
    }

    SubShader
    {
        Tags
        {
            "Queue" = "Transparent"
            "RenderType" = "Transparent"
            "RenderPipeline" = "UniversalPipeline"
            "IgnoreProjector" = "True"
        }

        Blend SrcAlpha OneMinusSrcAlpha
        Cull Off
        ZWrite Off

        Pass
        {
            Name "Universal2D"
            Tags { "LightMode" = "Universal2D" }

            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #pragma multi_compile USE_SHAPE_LIGHT_TYPE_0 __

            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            struct Attributes
            {
                float3 positionOS : POSITION;
                float4 color : COLOR;
                float2 uv : TEXCOORD0;
            };

            struct Varyings
            {
                float4 positionCS : SV_POSITION;
                float4 color : COLOR;
                float2 uv : TEXCOORD0;
                float2 lightingUV : TEXCOORD1;
            };

            TEXTURE2D(_MainTex);
            SAMPLER(sampler_MainTex);

            #if USE_SHAPE_LIGHT_TYPE_0
            TEXTURE2D(_ShapeLightTexture0);
            SAMPLER(sampler_ShapeLightTexture0);
            #endif

            CBUFFER_START(UnityPerMaterial)
                float4 _Color;
                float4 _MainTex_ST;
                float4 _RimColor;
                float _AmbientFloor;
                float _CelBands;
                float _EnableRim;
                float _RimThreshold;
                float _MetalThreshold;
                float _MetalIntensity;
            CBUFFER_END

            Varyings vert(Attributes input)
            {
                Varyings output = (Varyings)0;
                output.positionCS = TransformObjectToHClip(input.positionOS);
                output.uv = TRANSFORM_TEX(input.uv, _MainTex);
                output.color = input.color * _Color;

                float4 clipVertex = output.positionCS / output.positionCS.w;
                output.lightingUV = ComputeScreenPos(clipVertex).xy;
                return output;
            }

            half4 frag(Varyings input) : SV_Target
            {
                half4 mainTex = SAMPLE_TEXTURE2D(_MainTex, sampler_MainTex, input.uv);
                half4 baseCol = mainTex * input.color;

                #if USE_SHAPE_LIGHT_TYPE_0
                half4 light = SAMPLE_TEXTURE2D(_ShapeLightTexture0, sampler_ShapeLightTexture0, input.lightingUV);
                #else
                half4 light = half4(1, 1, 1, 1);
                #endif

                // 暗部下限锁定：防止弱受光区域像素变脏发黑
                light.rgb = max(light.rgb, half3(_AmbientFloor, _AmbientFloor, _AmbientFloor));

                // 赛璐珞光强阶梯量化
                half lightLum = dot(light.rgb, half3(0.299, 0.587, 0.114));
                half steppedLum = floor(lightLum * _CelBands) / _CelBands + (0.5 / _CelBands);
                half3 celLight = light.rgb * (steppedLum / max(lightLum, 0.001));

                half3 finalRgb = baseCol.rgb * celLight;

                // 顶端 UV 伪高光亮线
                if (_EnableRim > 0.5 && input.uv.y > _RimThreshold)
                {
                    finalRgb = lerp(finalRgb, _RimColor.rgb, 0.65 * baseCol.a);
                }

                // 金属亮度阈值阶梯反光
                half pixLum = dot(baseCol.rgb, half3(0.299, 0.587, 0.114));
                half metalMask = step(_MetalThreshold, pixLum) * _MetalIntensity;
                finalRgb = lerp(finalRgb, half3(1, 1, 1), metalMask);

                finalRgb *= baseCol.a;
                return half4(finalRgb, baseCol.a);
            }
            ENDHLSL
        }
    }
}
```

#### 产物 2：受光版地表流体着色器 `TileFlow.shader`
- **物理路径**：`Assets/Resources/materials/Shaders/TileFlow.shader`
- **设计意图**：全面替换原 Unlit 版 `TileFlow.shader`。完整保留原逐格相位噪声与水流扭曲算法，升级为 `Universal2D` Pass 并注入光照保底。

```hlsl
Shader "DeepseaOil/2D/TileFlow"
{
    Properties
    {
        [PerRendererData] _MainTex ("Sprite Texture", 2D) = "white" { }
        _Color ("Tint", Color) = (1, 1, 1, 1)
        _FlowSpeed ("Flow Speed", Float) = 3
        _FlowAmplitude ("Flow Amplitude", Range(0, 0.05)) = 0.012
        _FlowFrequency ("Flow Frequency", Float) = 20
        _PhaseNoise ("Phase Noise", Range(0, 1)) = 0 // 0=统一, 1=逐格随机
        _AmbientFloor ("Ambient Floor", Range(0.35, 0.6)) = 0.35
    }

    SubShader
    {
        Tags
        {
            "RenderType" = "Transparent"
            "Queue" = "Transparent"
            "RenderPipeline" = "UniversalPipeline"
            "IgnoreProjector" = "True"
        }

        Blend SrcAlpha OneMinusSrcAlpha
        ZWrite Off
        Cull Off

        Pass
        {
            Name "Universal2D"
            Tags { "LightMode" = "Universal2D" }

            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #pragma multi_compile USE_SHAPE_LIGHT_TYPE_0 __

            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            struct Attributes
            {
                float4 positionOS : POSITION;
                float4 color : COLOR;
                float2 uv : TEXCOORD0;
            };

            struct Varyings
            {
                float4 positionCS : SV_POSITION;
                float4 color : COLOR;
                float2 uv : TEXCOORD0;
                float2 worldPos : TEXCOORD1;
                float2 lightingUV : TEXCOORD2;
            };

            TEXTURE2D(_MainTex);
            SAMPLER(sampler_MainTex);

            #if USE_SHAPE_LIGHT_TYPE_0
            TEXTURE2D(_ShapeLightTexture0);
            SAMPLER(sampler_ShapeLightTexture0);
            #endif

            CBUFFER_START(UnityPerMaterial)
                float4 _Color;
                float4 _MainTex_ST;
                float _FlowSpeed;
                float _FlowAmplitude;
                float _FlowFrequency;
                float _PhaseNoise;
                float _AmbientFloor;
            CBUFFER_END

            Varyings vert(Attributes input)
            {
                Varyings output = (Varyings)0;
                output.positionCS = TransformObjectToHClip(input.positionOS.xyz);
                output.color = input.color * _Color;
                output.uv = TRANSFORM_TEX(input.uv, _MainTex);
                output.worldPos = TransformObjectToWorld(input.positionOS.xyz).xy;

                float4 clipVertex = output.positionCS / output.positionCS.w;
                output.lightingUV = ComputeScreenPos(clipVertex).xy;
                return output;
            }

            half4 frag(Varyings input) : SV_Target
            {
                float2 uv = input.uv;

                // 逐格相位量化计算
                float2 cellPos = floor(input.worldPos);
                float hash = frac(sin(dot(cellPos, float2(12.9898, 78.233))) * 43758.5453);
                float phase = hash * _PhaseNoise * 6.2831853;

                float dx = sin(_Time.y * _FlowSpeed + uv.y * _FlowFrequency + phase) * _FlowAmplitude;
                float dy = sin(_Time.y * _FlowSpeed * 1.3 + uv.x * _FlowFrequency * 0.7 + phase) * _FlowAmplitude;

                half4 baseCol = SAMPLE_TEXTURE2D(_MainTex, sampler_MainTex, uv);
                float lum = dot(baseCol.rgb, float3(0.299, 0.587, 0.114));
                float darkWeight = saturate(1.0 - lum);

                float2 flowUV = uv + float2(dx, dy) * darkWeight;
                half4 flowedColor = SAMPLE_TEXTURE2D(_MainTex, sampler_MainTex, flowUV) * input.color;

                #if USE_SHAPE_LIGHT_TYPE_0
                half4 light = SAMPLE_TEXTURE2D(_ShapeLightTexture0, sampler_ShapeLightTexture0, input.lightingUV);
                #else
                half4 light = half4(1, 1, 1, 1);
                #endif

                // 流水与地貌暗部保底
                light.rgb = max(light.rgb, half3(_AmbientFloor, _AmbientFloor, _AmbientFloor));

                half3 finalRgb = flowedColor.rgb * light.rgb;
                finalRgb *= flowedColor.a;

                return half4(finalRgb, flowedColor.a);
            }
            ENDHLSL
        }
    }
}
```

#### 产物 3：Editor 材质生成与 Tilemap 批量接线工具 `MaterialSetupTool.cs`
- **物理路径**：`Assets/Editor/Lighting/MaterialSetupTool.cs`
- **设计意图**：一键生成 `M_CelLit_Default.mat` 与 `M_TileFlow_Water.mat` 资产，并自动扫描替换 `Range.unity` 场景中 6 层 Tilemap 的材质引用，杜绝手工失误。

```csharp
#if UNITY_EDITOR
using UnityEditor;
using UnityEngine;
using UnityEngine.Tilemaps;

namespace DeepseaOil.EditorTools.Lighting
{
    /// <summary>第二批地表材质矩阵自动化配置工具</summary>
    public static class MaterialSetupTool
    {
        private const string MenuPath = "DeepseaOil/光照与后处理/一键装配地表受光材质矩阵";

        private const string CelLitShaderName = "DeepseaOil/2D/CelLit";
        private const string TileFlowShaderName = "DeepseaOil/2D/TileFlow";

        private const string CelLitMatPath = "Assets/Resources/materials/M_CelLit_Default.mat";
        private const string TileFlowMatPath = "Assets/Resources/materials/M_TileFlow_Water.mat";

        [MenuItem(MenuPath)]
        public static void SetupGroundMaterials()
        {
            Material celLitMat = EnsureMaterial(CelLitMatPath, CelLitShaderName);
            Material tileFlowMat = EnsureMaterial(TileFlowMatPath, TileFlowShaderName);

            ApplyToSceneTilemaps(celLitMat, tileFlowMat);

            AssetDatabase.SaveAssets();
            Debug.Log("[Material] 地表受光材质矩阵装配完成：Tilemap 材质已全部升级为受光版本。");
        }

        private static Material EnsureMaterial(string path, string shaderName)
        {
            Material mat = AssetDatabase.LoadAssetAtPath<Material>(path);
            Shader shader = Shader.Find(shaderName);

            if (shader == null)
            {
                Debug.LogError($"[Material] 未找到 Shader: {shaderName}，请检查 Shader 文件编译情况。");
                return null;
            }

            if (mat == null)
            {
                mat = new Material(shader);
                AssetDatabase.CreateAsset(mat, path);
            }
            else if (mat.shader != shader)
            {
                mat.shader = shader;
            }

            // 统一写入执行宪章要求的默认属性
            if (mat.HasProperty("_AmbientFloor")) mat.SetFloat("_AmbientFloor", 0.35f);
            if (mat.HasProperty("_CelBands")) mat.SetFloat("_CelBands", 3f);

            EditorUtility.SetDirty(mat);
            return mat;
        }

        private static void ApplyToSceneTilemaps(Material celLitMat, Material tileFlowMat)
        {
            TilemapRenderer[] renderers = Object.FindObjectsOfType<TilemapRenderer>(true);

            for (int i = 0; i < renderers.Length; i++)
            {
                TilemapRenderer tr = renderers[i];
                string goName = tr.gameObject.name;

                Undo.RecordObject(tr, "Upgrade Tilemap Material");

                // 水流层单独挂载 TileFlow 材质，其余 5 层挂载基础 CelLit
                if (goName.Contains("Water"))
                {
                    tr.sharedMaterial = tileFlowMat;
                }
                else
                {
                    tr.sharedMaterial = celLitMat;
                }

                EditorUtility.SetDirty(tr);
            }
        }
    }
}
#endif
```
