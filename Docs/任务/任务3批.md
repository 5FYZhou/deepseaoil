### 一、微观架构与动态光调度拓扑

为了在 0 GC 与严格控制 DrawCall 的前提下实现火、电、蒸汽的局部动态光源表现，架构采用**双事件驱动 + 对象池集中调度**模式：

```text
                     ┌──────────────────────────────┐
                     │          事件源广播          │
                     └──────────────┬───────────────┘
                                    │
           ┌────────────────────────┴────────────────────────┐
           ▼                                                 ▼
   【TileStateChanged】                              【GridBlastOccurred】
   火池/火海/电源/导电区                                 蒸汽爆轰瞬态
           │                                                 │
           ▼                                                 ▼
   ┌─────────────────────────────────────────────────────────────────┐
   │                  TileLightDirector (单向调度)                   │
   │  - 查表映射光照参数 (半径、色温、强度、调制种类)                │
   │  - 借用 / 归还 Light2D 对象池 (上限 maxSize: 32)                │
   │  - Update 批量推进 0 GC 呼吸与抽搐频闪                          │
   └────────────────────────────────┬────────────────────────────────┘
                                    │
                                    ▼
                      [Light2D 实例化对象]
                      - LightType = Point
                      - Target Sorting Layers = Ground + Actor 仅两层
```

### 二、工业级完整源码交付

#### 产物 1：动态点光源管理器 `TileLightDirector.cs`
- **物理路径**：`Assets/Scripts/Presentation/Lighting/TileLightDirector.cs`
- **设计意图**：统一接管全场元素地貌与爆轰事件的点光源生命周期。内置 32 容量 `Pool<GameObject>`，支持 2.5Hz 正弦呼吸火光、18Hz 阶跃抽搐电弧与 0.16s 瞬态衰减白光。

```csharp
using System.Collections.Generic;
using DeepseaOil.Foundation;
using DeepseaOil.Logic.Events;
using DeepseaOil.Logic.Grid;
using DeepseaOil.Presentation.Adapters;
using DeepseaOil.Presentation.Visual;
using UnityEngine;
using UnityEngine.Rendering.Universal;
using cfg.dso;

namespace DeepseaOil.Presentation.Lighting
{
    /// <summary>地块元素动态点光源管理器：管理火、电持续光及蒸汽爆轰瞬态白光</summary>
    /// <remarks>严格限制 32 盏上限，照射层级强制隔离在 Ground 与 Actor，避免 UI/特效合批打断。</remarks>
    [DisallowMultipleComponent]
    public sealed class TileLightDirector : MonoBehaviour
    {
        private enum LightKind
        {
            Fire = 0,
            Electric = 1,
            Steam = 2,
        }

        [Header("容量上限")]
        [Tooltip("最大并发点光源数量，超出部分直接丢弃")]
        [SerializeField] private int maxPoolSize = 32;

        [Header("可选接线（留空自动抓取）")]
        [SerializeField] private TilemapAdapter tilemapAdapter = default;

        private sealed class LightInstance
        {
            public GameObject Go;
            public Light2D Light;
            public LightKind Kind;
            public Vector3Int Cell;
            public float Elapsed;
            public float Duration;
            public float PhaseOffset;
        }

        private readonly Dictionary<Vector3Int, LightInstance> _activeContinuous = new();
        private readonly List<LightInstance> _activeTransients = new();
        private readonly List<Vector3Int> _recycleScratch = new();

        private Pool<GameObject> _pool;
        private GridGeometry _geometry;
        private bool _attached;
        private int _litLayersMask;

        public int ActiveLightCount => _activeContinuous.Count + _activeTransients.Count;

        private void Awake()
        {
            int groundId = SortingLayer.NameToID(SortingLayerConsts.Ground);
            int actorId = SortingLayer.NameToID(SortingLayerConsts.Actor);
            _litLayersMask = (1 << groundId) | (1 << actorId);

            _pool = new Pool<GameObject>(
                factory: CreateLightObject,
                onGet: go => go.SetActive(true),
                onRelease: go => go.SetActive(false),
                name: "TilePointLightPool",
                maxSize: maxPoolSize,
                overflowPolicy: PoolOverflowPolicy.CreateOrDrop,
                onDestroy: Destroy);
        }

        private void Start()
        {
            Attach();
        }

        public void Attach()
        {
            if (_attached) return;

            if (tilemapAdapter == null)
            {
                tilemapAdapter = FindObjectOfType<TilemapAdapter>();
            }

            if (tilemapAdapter != null)
            {
                _geometry = tilemapAdapter.ReadGeometry();
            }

            EventBus<TileStateChanged>.Subscribe(OnTileStateChanged);
            EventBus<GridBlastOccurred>.Subscribe(OnGridBlastOccurred);
            _attached = true;
        }

        public void Detach()
        {
            if (!_attached) return;

            EventBus<TileStateChanged>.Unsubscribe(OnTileStateChanged);
            EventBus<GridBlastOccurred>.Unsubscribe(OnGridBlastOccurred);
            _attached = false;

            ClearAll();
        }

        private void OnDestroy()
        {
            Detach();
            _pool?.Dispose();
            _pool = null;
        }

        private void Update()
        {
            float dt = Time.deltaTime;
            float time = Time.time;

            // 1. 持续型光源调制（火池 2.5Hz 正弦呼吸 / 电源 18Hz 阶跃电弧）
            foreach (KeyValuePair<Vector3Int, LightInstance> kv in _activeContinuous)
            {
                LightInstance inst = kv.Value;

                if (inst.Kind == LightKind.Fire)
                {
                    // 2.5 Hz 呼吸：周期约 0.4s，角速度 15.70796
                    float wave = (Mathf.Sin(time * 15.70796f + inst.PhaseOffset) + 1f) * 0.5f;
                    inst.Light.intensity = Mathf.Lerp(0.85f, 1.05f, wave);
                }
                else if (inst.Kind == LightKind.Electric)
                {
                    // 18 Hz 抽搐阶跃：使用时间切片生成非连续高频阶梯
                    float step = Mathf.Floor(time * 18f + inst.PhaseOffset);
                    float hash = Mathf.Repeat(step * 0.6180339f, 1f);
                    inst.Light.intensity = Mathf.Lerp(0.90f, 1.20f, hash);
                }
            }

            // 2. 瞬态型光源衰减与到期回收
            for (int i = _activeTransients.Count - 1; i >= 0; i--)
            {
                LightInstance inst = _activeTransients[i];
                inst.Elapsed += dt;

                if (inst.Elapsed >= inst.Duration)
                {
                    RecycleInstance(inst);
                    _activeTransients.RemoveAt(i);
                    continue;
                }

                // 瞬态线性归零
                float progress = inst.Elapsed / inst.Duration;
                inst.Light.intensity = (1f - progress) * 1.0f;
            }
        }

        public void ClearAll()
        {
            foreach (KeyValuePair<Vector3Int, LightInstance> kv in _activeContinuous)
            {
                RecycleInstance(kv.Value);
            }
            _activeContinuous.Clear();

            for (int i = 0; i < _activeTransients.Count; i++)
            {
                RecycleInstance(_activeTransients[i]);
            }
            _activeTransients.Clear();
        }

        private void OnTileStateChanged(TileStateChanged evt)
        {
            // 先归还本格旧光源（无论新状态是否有光）
            if (_activeContinuous.TryGetValue(evt.Cell, out LightInstance oldInst))
            {
                RecycleInstance(oldInst);
                _activeContinuous.Remove(evt.Cell);
            }

            if (!TryGetContinuousLightSpec(evt.State, out LightKind kind, out Color color, out float radius))
            {
                return;
            }

            if (!_geometry.IsValid) return;

            if (!_pool.TryGet(out GameObject obj) || obj == null) return;

            Vector2 center = _geometry.CellCenter(evt.Cell);
            obj.transform.position = new Vector3(center.x, center.y, 0f);

            Light2D light2d = obj.GetComponent<Light2D>();
            light2d.color = color;
            light2d.pointLightOuterRadius = radius;
            light2d.pointLightInnerRadius = radius * 0.2f;

            var inst = new LightInstance
            {
                Go = obj,
                Light = light2d,
                Kind = kind,
                Cell = evt.Cell,
                PhaseOffset = (evt.Cell.x * 17 + evt.Cell.y * 31) % 100 * 0.1f,
            };

            _activeContinuous[evt.Cell] = inst;
        }

        private void OnGridBlastOccurred(GridBlastOccurred evt)
        {
            if (evt.ResultTile != TileStateType.Steam) return;

            if (!_pool.TryGet(out GameObject obj) || obj == null) return;

            obj.transform.position = new Vector3(evt.Center.x, evt.Center.y, 0f);

            Light2D light2d = obj.GetComponent<Light2D>();
            light2d.color = Color.white;
            light2d.pointLightOuterRadius = 2.2f;
            light2d.pointLightInnerRadius = 0.5f;
            light2d.intensity = 1.0f;

            var inst = new LightInstance
            {
                Go = obj,
                Light = light2d,
                Kind = LightKind.Steam,
                Elapsed = 0f,
                Duration = 0.16f,
            };

            _activeTransients.Add(inst);
        }

        private static bool TryGetContinuousLightSpec(TileStateType state, out LightKind kind, out Color color, out float radius)
        {
            switch (state)
            {
                case TileStateType.BasicFire:
                case TileStateType.FlameField:
                    kind = LightKind.Fire;
                    color = new Color(1.0f, 0.549f, 0.259f, 1f); // #FF8C42 暖橙红
                    radius = 1.8f;
                    return true;

                case TileStateType.BasicElectricity:
                case TileStateType.ConductZone:
                case TileStateType.ChargedThorn:
                    kind = LightKind.Electric;
                    color = new Color(0.490f, 0.976f, 1.0f, 1f); // #7DF9FF 高亮青白
                    radius = 1.5f;
                    return true;

                default:
                    kind = default;
                    color = default;
                    radius = 0f;
                    return false;
            }
        }

        private GameObject CreateLightObject()
        {
            var go = new GameObject("TilePointLight");
            go.transform.SetParent(transform, false);

            var light2d = go.AddComponent<Light2D>();
            light2d.lightType = Light2D.LightType.Point;

            // 严格过滤：仅照射 Ground 与 Actor 层
            int groundId = SortingLayer.NameToID(SortingLayerConsts.Ground);
            int actorId = SortingLayer.NameToID(SortingLayerConsts.Actor);
            light2d.applyToSortingLayers = new[] { groundId, actorId };

            go.SetActive(false);
            return go;
        }

        private void RecycleInstance(LightInstance inst)
        {
            if (inst.Go != null && _pool != null)
            {
                _pool.Release(inst.Go);
            }
        }
    }
}
```

#### 产物 2：Editor 菜单一键装配工具 `PointLightSetupTool.cs`
- **物理路径**：`Assets/Editor/Lighting/PointLightSetupTool.cs`
- **设计意图**：一键在场景 `[Environment]` 节点下挂载并装配 `TileLightDirector`，自动建立关联。

```csharp
#if UNITY_EDITOR
using DeepseaOil.Presentation.Adapters;
using DeepseaOil.Presentation.Lighting;
using UnityEditor;
using UnityEngine;

namespace DeepseaOil.EditorTools.Lighting
{
    /// <summary>第三批点光源管理器装配工具</summary>
    public static class PointLightSetupTool
    {
        private const string MenuPath = "DeepseaOil/光照与后处理/一键装配动态点光源系统";

        [MenuItem(MenuPath)]
        public static void SetupTileLightDirector()
        {
            GameObject root = GameObject.Find("[Environment]");
            if (root == null)
            {
                root = new GameObject("[Environment]");
                Undo.RegisterCreatedObjectUndo(root, "Create [Environment]");
            }

            TileLightDirector director = root.GetComponent<TileLightDirector>();
            if (director == null)
            {
                director = root.AddComponent<TileLightDirector>();
                Undo.RegisterCreatedObjectUndo(director, "Add TileLightDirector");
            }

            TilemapAdapter adapter = Object.FindObjectOfType<TilemapAdapter>();
            if (adapter != null)
            {
                SerializedObject so = new SerializedObject(director);
                so.FindProperty("tilemapAdapter").objectReferenceValue = adapter;
                so.ApplyModifiedProperties();
            }

            EditorUtility.SetDirty(director);
            Debug.Log("[Lighting] 动态点光源系统已就位：TileLightDirector 挂载成功，已接线 TilemapAdapter。");
        }
    }
}
#endif
```

using DeepseaOil.Presentation.Lighting;
using DeepseaOil.Presentation.Visual;
using NUnit.Framework;
using UnityEngine;
using UnityEngine.Rendering.Universal;

namespace DeepseaOil.Tests.EditMode
{
    [TestFixture]
    public sealed class TileLightDirectorTests
    {
        [Test]
        public void 点光源挂载_TargetSortingLayers严格仅包含Ground与Actor()
        {
            var go = new GameObject("TestLight");
            var light2d = go.AddComponent<Light2D>();
            light2d.lightType = Light2D.LightType.Point;

            int groundId = SortingLayer.NameToID(SortingLayerConsts.Ground);
            int actorId = SortingLayer.NameToID(SortingLayerConsts.Actor);
            light2d.applyToSortingLayers = new[] { groundId, actorId };

            int[] applied = light2d.applyToSortingLayers;
            Assert.AreEqual(2, applied.Length, "点光源照射的 Sorting Layer 数量不是 2 个！");

            bool hasGround = false;
            bool hasActor = false;
            for (int i = 0; i < applied.Length; i++)
            {
                if (applied[i] == groundId) hasGround = true;
                if (applied[i] == actorId) hasActor = true;
            }

            Assert.IsTrue(hasGround && hasActor, "点光源漏选了 Ground 或 Actor 层！");

            Object.DestroyImmediate(go);
        }

        [Test]
        public void 容量防线_初始未激活光源数为零()
        {
            var go = new GameObject("TestDirector");
            var director = go.AddComponent<TileLightDirector>();

            Assert.AreEqual(0, director.ActiveLightCount, "未开始驱动时活动光源数不为 0！");

            Object.DestroyImmediate(go);
        }
    }
}

### 六、交付验收清单 (DoD 自查表)

- [x] **架构层**：严格通过 `EventBus<TileStateChanged>` 与 `EventBus<GridBlastOccurred>` 监听，纯表现层解耦，Logic 层零反向依赖。
