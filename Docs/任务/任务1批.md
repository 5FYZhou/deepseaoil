### 一、架构拓扑与层级隔离方案

#### 1. Sorting Layer 物理分层与受光过滤矩阵

要保证“地貌/角色受光，特效/UI 绝对自发光（Unlit）”，必须在 `TagManager` 建立物理图层隔离。2D 光源（`Light2D`）的 `Target Sorting Layers` 只能按 Layer 过滤，无法按 Order 过滤。

```text
  [渲染从底到顶 (从早到晚)]
  Layer 0: Default        ← 空置保留（引擎兜底）
  Layer 1: Ground         ← 受光 (Tilemap 地表层与底纹)
  Layer 2: Actor          ← 受光 (角色真身、影子 310、立面覆盖物、Y-Sort 频带 400~1000)
  Layer 3: Aim            ← 不受光 (100% Unlit，瞄准高亮框 320)
  Layer 4: Effect         ← 不受光 (100% Unlit，粒子微粒、弹幕、飘字、碎片 1200)
  Layer 5: UI             ← 不受光 (100% Unlit，HUD、全屏裂纹、弹窗)
```

| Sorting Layer 名 | 包含内容                       | 默认 Order 频带 | 受 Global Light 2D 照射 | 选用 Shader 模型                  |
| :--------------- | :----------------------------- | :-------------- | :---------------------- | :-------------------------------- |
| **Default**      | 引擎默认空槽                   | 0               | 否                      | -                                 |
| **Ground**       | 6 层 Tilemap 地面与底纹        | 0 ~ 5           | **是 (开启)**           | Cel-Lit（第二批切）/ Sprite-Lit   |
| **Actor**        | 玩家、敌人、影子件、立面装饰   | 310, 400 ~ 1000 | **是 (开启)**           | Cel-Lit / SpriteFlash（材质插值） |
| **Aim**          | 瞄准落点反馈高亮               | 320             | **否 (阻断)**           | Unlit                             |
| **Effect**       | 粒子特效、受击火花、飘字、碎片 | 1200+           | **否 (阻断)**           | Unlit（粒子与文字固有色）         |
| **UI**           | HUD 读数、菜单、全屏裂纹       | 0 ~ 100         | **否 (阻断)**           | UI/Default (Unlit)                |

namespace DeepseaOil.Presentation.Lighting

        Dusk = 1,
        Night = 2,

#### 产物 3：全局光照时段控制器 `EnvironmentDirector.cs`
- **物理路径**：`Assets/Scripts/Presentation/Lighting/EnvironmentDirector.cs`
- **设计意图**：单向驱动场景中的 `Light2D` 组件，提供无 GC 的时段切换与平滑插值通道。

```csharp
using UnityEngine;
using UnityEngine.Rendering.Universal;

namespace DeepseaOil.Presentation.Lighting
{
    /// <summary>全局环境光照与时段表现控制器</summary>
    /// <remarks>驱动场景唯一 Global Light 2D，插值计算零堆分配，未绑定光源时安全降级。</remarks>
    [DisallowMultipleComponent]
    public sealed class EnvironmentDirector : MonoBehaviour
    {
        [Header("光照引用")]
        [Tooltip("场景主全局光照")]
        [SerializeField] private Light2D globalLight = default;

        [Header("初始配置")]
        [SerializeField] private TimeOfDayKind initialTimeOfDay = TimeOfDayKind.Day;

        [Tooltip("时段过渡插值时长（秒）")]
        [SerializeField] private float transitionDuration = 1.5f;

        private TimeOfDaySetting _currentSetting;
        private TimeOfDaySetting _targetSetting;

        private float _transitionTime;
        private bool _isTransitioning;

        public TimeOfDayKind CurrentTimeOfDay => _targetSetting.Kind;

        private void Awake()
        {
            if (globalLight == null)
            {
                globalLight = GetComponentInChildren<Light2D>();
            }

            _currentSetting = TimeOfDaySetting.GetDefault(initialTimeOfDay);
            _targetSetting = _currentSetting;

            ApplySettingDirectly(in _currentSetting);
        }

        private void Update()
        {
            if (!_isTransitioning) return;

            _transitionTime += Time.deltaTime;
            float t = Mathf.Clamp01(_transitionTime / transitionDuration);

            // 赛璐珞式平滑过渡：使用平滑缓动曲线消除机械顿挫感
            float smoothT = Mathf.SmoothStep(0f, 1f, t);

            if (globalLight != null)
            {
                globalLight.color = Color.Lerp(_currentSetting.LightColor, _targetSetting.LightColor, smoothT);
                globalLight.intensity = Mathf.Lerp(_currentSetting.Intensity, _targetSetting.Intensity, smoothT);
            }

            if (t >= 1f)
            {
                _isTransitioning = false;
                _currentSetting = _targetSetting;
                ApplySettingDirectly(in _currentSetting);
            }
        }

        /// <summary>平滑切换至目标时段</summary>
        public void TransitionTo(TimeOfDayKind kind)
        {
            if (_targetSetting.Kind == kind && !_isTransitioning) return;

            _targetSetting = TimeOfDaySetting.GetDefault(kind);
            _transitionTime = 0f;
            _isTransitioning = true;
        }

        /// <summary>瞬时设置时段（无过渡）</summary>
        public void SetTimeOfDayImmediate(TimeOfDayKind kind)
        {
            _isTransitioning = false;
            _currentSetting = TimeOfDaySetting.GetDefault(kind);
            _targetSetting = _currentSetting;

            ApplySettingDirectly(in _currentSetting);
        }

        private void ApplySettingDirectly(in TimeOfDaySetting setting)
        {
            if (globalLight == null)
            {
                Debug.LogWarning("[Lighting] EnvironmentDirector 未绑定 Global Light 2D 组件，光照指令未生效。", this);
                return;
            }

            globalLight.color = setting.LightColor;
            globalLight.intensity = setting.Intensity;
        }
    }
}
```

#### 产物 4：Editor 自动化装配与封印验证工具 `EnvironmentSetupTool.cs`
- **物理路径**：`Assets/Editor/Lighting/EnvironmentSetupTool.cs`
- **设计意图**：一键在当前场景完成 Global Light 2D 放置、图层受光过滤配置、Global Volume 封印参数注入与层级自检。

```csharp
#if UNITY_EDITOR
using DeepseaOil.Presentation.Lighting;
using DeepseaOil.Presentation.Visual;
using UnityEditor;
using UnityEngine;
using UnityEngine.Rendering;
using UnityEngine.Rendering.Universal;

namespace DeepseaOil.EditorTools.Lighting
{
    /// <summary>全局光照与后处理环境一键装配工具</summary>
    public static class EnvironmentSetupTool
    {
        private const string MenuPath = "DeepseaOil/光照与后处理/一键装配当前场景环境基座";
        private const string ProfileAssetPath = "Assets/Settings/DefaultGlobalVolumeProfile.asset";

        [MenuItem(MenuPath)]
        public static void SetupSceneEnvironment()
        {
            SetupSortingLayers();

            GameObject root = GameObject.Find("[Environment]");
            if (root == null)
            {
                root = new GameObject("[Environment]");
                Undo.RegisterCreatedObjectUndo(root, "Create [Environment] Root");
            }

            Light2D globalLight = SetupGlobalLight(root);
            SetupGlobalVolume(root);
            SetupEnvironmentDirector(root, globalLight);

            EditorUtility.SetDirty(root);
            Debug.Log("[Lighting] 环境基座装配完成：Global Light 2D、Global Volume、EnvironmentDirector 已就位并封印。");
        }

        private static void SetupSortingLayers()
        {
            SerializedObject tagManager = new SerializedObject(AssetDatabase.LoadAllAssetsAtPath("ProjectSettings/TagManager.asset")[0]);
            SerializedProperty sortingLayers = tagManager.FindProperty("m_SortingLayers");

            EnsureSortingLayer(sortingLayers, SortingLayerConsts.Ground);
            EnsureSortingLayer(sortingLayers, SortingLayerConsts.Actor);
            EnsureSortingLayer(sortingLayers, SortingLayerConsts.Aim);
            EnsureSortingLayer(sortingLayers, SortingLayerConsts.Effect);
            EnsureSortingLayer(sortingLayers, SortingLayerConsts.UI);

            tagManager.ApplyModifiedProperties();
        }

        private static void EnsureSortingLayer(SerializedProperty sortingLayers, string layerName)
        {
            for (int i = 0; i < sortingLayers.arraySize; i++)
            {
                SerializedProperty element = sortingLayers.GetArrayElementAtIndex(i);
                SerializedProperty nameProp = element.FindPropertyRelative("name");
                if (nameProp.stringValue == layerName) return;
            }

            int index = sortingLayers.arraySize;
            sortingLayers.InsertArrayElementAtIndex(index);
            SerializedProperty newElement = sortingLayers.GetArrayElementAtIndex(index);
            newElement.FindPropertyRelative("name").stringValue = layerName;
            newElement.FindPropertyRelative("uniqueID").intValue = checked((int)MurmurHash(layerName));
        }

        private static uint MurmurHash(string str)
        {
            uint hash = 0x811c9dc5;
            for (int i = 0; i < str.Length; i++)
            {
                hash ^= str[i];
                hash *= 0x01000193;
            }
            return hash;
        }

        private static Light2D SetupGlobalLight(GameObject root)
        {
            Transform lightTrans = root.transform.Find("Global Light 2D");
            GameObject lightGo = lightTrans != null ? lightTrans.gameObject : null;

            if (lightGo == null)
            {
                lightGo = new GameObject("Global Light 2D");
                lightGo.transform.SetParent(root.transform, false);
                Undo.RegisterCreatedObjectUndo(lightGo, "Create Global Light 2D");
            }

            Light2D light2d = lightGo.GetComponent<Light2D>();
            if (light2d == null) light2d = lightGo.AddComponent<Light2D>();

            light2d.lightType = Light2D.LightType.Global;

            TimeOfDaySetting daySetting = TimeOfDaySetting.GetDefault(TimeOfDayKind.Day);
            light2d.color = daySetting.LightColor;
            light2d.intensity = daySetting.Intensity;

            // 严格过滤：仅勾选 Ground 与 Actor 两个层
            int groundLayerId = SortingLayer.NameToID(SortingLayerConsts.Ground);
            int actorLayerId = SortingLayer.NameToID(SortingLayerConsts.Actor);
            light2d.applyToSortingLayers = new[] { groundLayerId, actorLayerId };

            return light2d;
        }

        private static void SetupGlobalVolume(GameObject root)
        {
            Transform volumeTrans = root.transform.Find("Global Volume");
            GameObject volumeGo = volumeTrans != null ? volumeTrans.gameObject : null;

            if (volumeGo == null)
            {
                volumeGo = new GameObject("Global Volume");
                volumeGo.transform.SetParent(root.transform, false);
                Undo.RegisterCreatedObjectUndo(volumeGo, "Create Global Volume");
            }

            Volume volume = volumeGo.GetComponent<Volume>();
            if (volume == null) volume = volumeGo.AddComponent<Volume>();

            volume.isGlobal = true;

            VolumeProfile profile = AssetDatabase.LoadAssetAtPath<VolumeProfile>(ProfileAssetPath);
            if (profile == null)
            {
                profile = ScriptableObject.CreateInstance<VolumeProfile>();
                AssetDatabase.CreateAsset(profile, ProfileAssetPath);
                AssetDatabase.SaveAssets();
            }

            ConfigureSealedOverrides(profile);
            volume.sharedProfile = profile;
        }

        private static void ConfigureSealedOverrides(VolumeProfile profile)
        {
            // 1. Bloom 封印参数
            if (!profile.TryGet(out Bloom bloom)) bloom = profile.Add<Bloom>(true);
            bloom.threshold.overrideState = true;
            bloom.threshold.value = 1.15f; // 核心保险：大于 1.0 防止受击闪白变大白斑
            bloom.intensity.overrideState = true;
            bloom.intensity.value = 0.45f;
            bloom.scatter.overrideState = true;
            bloom.scatter.value = 0.60f;

            // 2. Vignette 暗角封印参数
            if (!profile.TryGet(out Vignette vignette)) vignette = profile.Add<Vignette>(true);
            vignette.color.overrideState = true;
            vignette.color.value = new Color(0.118f, 0.118f, 0.114f, 1f); // #1E1E1D 深灰
            vignette.intensity.overrideState = true;
            vignette.intensity.value = 0.22f;
            vignette.smoothness.overrideState = true;
            vignette.smoothness.value = 0.40f;

            // 3. Tonemapping 色调映射封印参数
            if (!profile.TryGet(out Tonemapping tonemapping)) tonemapping = profile.Add<Tonemapping>(true);
            tonemapping.mode.overrideState = true;
            tonemapping.mode.value = TonemappingMode.Neutral; // 坚决不使用 ACES

            EditorUtility.SetDirty(profile);
        }

        private static void SetupEnvironmentDirector(GameObject root, Light2D globalLight)
        {
            EnvironmentDirector director = root.GetComponent<EnvironmentDirector>();
            if (director == null) director = root.AddComponent<EnvironmentDirector>();

            SerializedObject serializedDirector = new SerializedObject(director);
            SerializedProperty propLight = serializedDirector.FindProperty("globalLight");
            propLight.objectReferenceValue = globalLight;
            serializedDirector.ApplyModifiedProperties();
        }
    }
}
#endif
```
