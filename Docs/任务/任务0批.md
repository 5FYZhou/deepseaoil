### 一、第 0 批核心定位与零破坏准则

#### 1. 为什么必须先执行「第 0 批（纯地基工作）」？
在以往的模块顺序执行中，若直接上手修改场景或挂载组件，往往会因为**底层的图层常量缺失、Shader 尚未通过编译、或者枚举契约未就位**而产生连环编译红字（CS0103 / CS0246），甚至导致 Prefab 上的材质丢失变成粉色。

**第 0 批的第一性原则：**
- **只铺契约与资源，不改运行接线**：引入所有后续依赖的常量、枚举、物理参数结构体与 Shader 资产。
- **即刻可编译、零行为回归**：交付后立即执行 `pwsh Tools/compile-gate.ps1`，必须 0 error / 0 warning。
- **为后续批次提供强类型支撑**：第 1 批（场景基座）、第 2 批（材质矩阵替换）、第 3 批（点光源调度）全部基于第 0 批的确定性契约推进，杜绝任何空中楼阁式的开发。

#### 2. 第 0 批产物全景表

#### 产物 1：排序图层常量与安全契约 `SortingLayerConsts.cs`
- **物理路径**：`Assets/Scripts/Presentation/Visual/SortingLayerConsts.cs`
- **设计意图**：提供强类型图层名称。**使用动态只读属性替代 `static readonly` 字段**，彻底规避因 Domain Reload 顺序导致图层 ID 被错误缓存为 0 的引擎底层隐患。

```csharp
using UnityEngine;

namespace DeepseaOil.Presentation.Visual
{
    /// <summary>渲染排序图层常量契约与动态安全 ID 访问器</summary>
    public static class SortingLayerConsts
    {
        public const string Default = "Default";
        public const string Ground = "Ground";
        public const string Actor = "Actor";
        public const string Aim = "Aim";
        public const string Effect = "Effect";
        public const string UI = "UI";

        public static int GroundId => SortingLayer.NameToID(Ground);
        public static int ActorId => SortingLayer.NameToID(Actor);
        public static int AimId => SortingLayer.NameToID(Aim);
        public static int EffectId => SortingLayer.NameToID(Effect);
        public static int UIId => SortingLayer.NameToID(UI);

        /// <summary>受光物理掩码：仅 Ground 与 Actor 允许接收光照计算</summary>
        public static int LitLayersMask
        {
            get
            {
                int groundId = GroundId;
                int actorId = ActorId;
                return (groundId >= 0 ? 1 << groundId : 0) | (actorId >= 0 ? 1 << actorId : 0);
            }
        }
    }
}
```

#### 产物 2：时段定义与配置值类型 `TimeOfDay.cs`
- **物理路径**：`Assets/Scripts/Presentation/Lighting/TimeOfDay.cs`
- **设计意图**：纯值类型，将执行宪章中白天（0.75）、黄昏（0.60）、夜晚（0.45）的物理参数与暗部补偿固化，供后续环境控制器安全调度。

```csharp
using System;
using UnityEngine;

namespace DeepseaOil.Presentation.Lighting
{
    /// <summary>全局时段枚举</summary>
    public enum TimeOfDayKind
    {
        Day = 0,
        Dusk = 1,
        Night = 2,
    }

    /// <summary>时段物理参数封装，环境光强度严格遵守暗部保底大于等于 0.35</summary>
    [Serializable]
    public struct TimeOfDaySetting
    {
        public TimeOfDayKind Kind;

        [Tooltip("全局主光颜色")]
        public Color LightColor;

        [Tooltip("全局光照强度，白天 0.75，黄昏 0.60，夜晚 0.45")]
        [Range(0.35f, 1.2f)]
        public float Intensity;

        [Tooltip("暗部环境补偿色，用于防止冷色调跌入死黑")]
        public Color AmbientFloorColor;

        public static TimeOfDaySetting GetDefault(TimeOfDayKind kind)
        {
            switch (kind)
            {
                case TimeOfDayKind.Dusk:
                    return new TimeOfDaySetting
                    {
                        Kind = TimeOfDayKind.Dusk,
                        LightColor = new Color(0.973f, 0.702f, 0.580f, 1f), // #F8B394 粉橘色
                        Intensity = 0.60f,
                        AmbientFloorColor = new Color(0.227f, 0.165f, 0.220f, 1f) // #3A2A38 深粉紫
                    };

                case TimeOfDayKind.Night:
                    return new TimeOfDaySetting
                    {
                        Kind = TimeOfDayKind.Night,
                        LightColor = new Color(0.478f, 0.459f, 0.627f, 1f), // #7A75A0 蓝紫色
                        Intensity = 0.45f,
                        AmbientFloorColor = new Color(0.133f, 0.125f, 0.208f, 1f) // #222035 深蓝灰
                    };

                default:
                    return new TimeOfDaySetting
                    {
                        Kind = TimeOfDayKind.Day,
                        LightColor = new Color(1f, 0.961f, 0.867f, 1f), // #FFF5DD 暖日光
                        Intensity = 0.75f,
                        AmbientFloorColor = new Color(0.35f, 0.35f, 0.35f, 1f)
                    };
            }
        }
    }
}
```

#### 产物 3：赛璐珞受光与受击闪白统一着色器 `CelLit.shader`
- **物理路径**：`Assets/Resources/materials/Shaders/CelLit.shader`
- **设计意图**：地表与角色通用的核心受光着色器。融合了 0.35 暗部截断、3 阶赛璐珞阶梯、顶端 UV 亮线、金属硬高光，以及 `_FlashAmount` 纯白剪影材质插值，彻底解开受光与闪白的死锁。

```hlsl
Shader "DeepseaOil/2D/CelLit"
{
    Properties
    {
        [PerRendererData] _MainTex ("Sprite Texture", 2D) = "white" {}
        _Color ("Tint", Color) = (1, 1, 1, 1)

        [Header(Cel Shading and Floor)]
        _AmbientFloor ("Ambient Floor", Range(0.35, 0.6)) = 0.35
        _CelBands ("Cel Bands Count", Range(2, 4)) = 3

        [Header(Rim and Metal Highlight)]
        [Toggle] _EnableRim ("Enable Top Rim Highlight", Float) = 0
        _RimThreshold ("Rim Threshold UV Y", Range(0.8, 1.0)) = 0.92
        _RimColor ("Rim Highlight Color", Color) = (0.976, 0.808, 0.353, 1) // #F9CE5A 明黄
        _MetalThreshold ("Metal Highlight Threshold", Range(0.7, 0.95)) = 0.85
        _MetalIntensity ("Metal Highlight Intensity", Range(0.0, 1.0)) = 0.0

        [Header(Hit Flash Integration)]
        _FlashAmount ("Flash Amount", Range(0.0, 1.0)) = 0.0
        _FlashColor ("Flash Color", Color) = (1, 1, 1, 1)
    }

    SubShader
    {
        Tags
        {
            "Queue" = "Transparent"
            "RenderType" = "Transparent"
            "RenderPipeline" = "UniversalPipeline"
            "IgnoreProjector" = "True"
        }

        Blend SrcAlpha OneMinusSrcAlpha
        Cull Off
        ZWrite Off

        Pass
        {
            Name "Universal2D"
            Tags { "LightMode" = "Universal2D" }

            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #pragma multi_compile USE_SHAPE_LIGHT_TYPE_0 __

            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            struct Attributes
            {
                float3 positionOS : POSITION;
                float4 color : COLOR;
                float2 uv : TEXCOORD0;
            };

            struct Varyings
            {
                float4 positionCS : SV_POSITION;
                float4 color : COLOR;
                float2 uv : TEXCOORD0;
                float4 screenUV : TEXCOORD1;
            };

            TEXTURE2D(_MainTex);
            SAMPLER(sampler_MainTex);

            #if USE_SHAPE_LIGHT_TYPE_0
            TEXTURE2D(_ShapeLightTexture0);
            SAMPLER(sampler_ShapeLightTexture0);
            #endif

            CBUFFER_START(UnityPerMaterial)
                float4 _Color;
                float4 _MainTex_ST;
                float4 _RimColor;
                float4 _FlashColor;
                float _FlashAmount;
                float _AmbientFloor;
                float _CelBands;
                float _EnableRim;
                float _RimThreshold;
                float _MetalThreshold;
                float _MetalIntensity;
            CBUFFER_END

            Varyings vert(Attributes input)
            {
                Varyings output = (Varyings)0;
                output.positionCS = TransformObjectToHClip(input.positionOS);
                output.uv = TRANSFORM_TEX(input.uv, _MainTex);
                output.color = input.color * _Color;
                output.screenUV = ComputeScreenPos(output.positionCS);
                return output;
            }

            half4 frag(Varyings input) : SV_Target
            {
                half4 mainTex = SAMPLE_TEXTURE2D(_MainTex, sampler_MainTex, input.uv);
                half4 baseCol = mainTex * input.color;

                #if USE_SHAPE_LIGHT_TYPE_0
                float2 lightUV = input.screenUV.xy / input.screenUV.w;
                half4 light = SAMPLE_TEXTURE2D(_ShapeLightTexture0, sampler_ShapeLightTexture0, lightUV);
                #else
                half4 light = half4(1, 1, 1, 1);
                #endif

                light.rgb = max(light.rgb, half3(_AmbientFloor, _AmbientFloor, _AmbientFloor));

                half lightLum = dot(light.rgb, half3(0.299, 0.587, 0.114));
                half steppedLum = floor(lightLum * _CelBands) / _CelBands + (0.5 / _CelBands);
                half3 celLight = light.rgb * (steppedLum / max(lightLum, 0.001));

                half3 finalRgb = baseCol.rgb * celLight;

                if (_EnableRim > 0.5 && input.uv.y > _RimThreshold)
                {
                    finalRgb = lerp(finalRgb, _RimColor.rgb, 0.65 * baseCol.a);
                }

                half pixLum = dot(baseCol.rgb, half3(0.299, 0.587, 0.114));
                half metalMask = step(_MetalThreshold, pixLum) * _MetalIntensity;
                finalRgb = lerp(finalRgb, half3(1, 1, 1), metalMask);

                finalRgb = lerp(finalRgb, _FlashColor.rgb, _FlashAmount);

                finalRgb *= baseCol.a;
                return half4(finalRgb, baseCol.a);
            }
            ENDHLSL
        }
    }
}
```

#### 产物 4：受光流体着色器底座 `TileFlow.shader`
- **物理路径**：`Assets/Resources/materials/Shaders/TileFlow.shader`
- **设计意图**：修正正交屏幕空间 UV 计算，升级为支持 2D 光照纹理采样的 Universal2D Pass，同时保留水流流动与逐格相位噪声逻辑。

```hlsl
Shader "DeepseaOil/2D/TileFlow"
{
    Properties
    {
        [PerRendererData] _MainTex ("Sprite Texture", 2D) = "white" { }
        _Color ("Tint", Color) = (1, 1, 1, 1)
        _FlowSpeed ("Flow Speed", Float) = 3
        _FlowAmplitude ("Flow Amplitude", Range(0, 0.05)) = 0.012
        _FlowFrequency ("Flow Frequency", Float) = 20
        _PhaseNoise ("Phase Noise", Range(0, 1)) = 0
        _AmbientFloor ("Ambient Floor", Range(0.35, 0.6)) = 0.35
    }

    SubShader
    {
        Tags
        {
            "RenderType" = "Transparent"
            "Queue" = "Transparent"
            "RenderPipeline" = "UniversalPipeline"
            "IgnoreProjector" = "True"
        }

        Blend SrcAlpha OneMinusSrcAlpha
        ZWrite Off
        Cull Off

        Pass
        {
            Name "Universal2D"
            Tags { "LightMode" = "Universal2D" }

            HLSLPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #pragma multi_compile USE_SHAPE_LIGHT_TYPE_0 __

            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"

            struct Attributes
            {
                float4 positionOS : POSITION;
                float4 color : COLOR;
                float2 uv : TEXCOORD0;
            };

            struct Varyings
            {
                float4 positionCS : SV_POSITION;
                float4 color : COLOR;
                float2 uv : TEXCOORD0;
                float2 worldPos : TEXCOORD1;
                float4 screenUV : TEXCOORD2;
            };

            TEXTURE2D(_MainTex);
            SAMPLER(sampler_MainTex);

            #if USE_SHAPE_LIGHT_TYPE_0
            TEXTURE2D(_ShapeLightTexture0);
            SAMPLER(sampler_ShapeLightTexture0);
            #endif

            CBUFFER_START(UnityPerMaterial)
                float4 _Color;
                float4 _MainTex_ST;
                float _FlowSpeed;
                float _FlowAmplitude;
                float _FlowFrequency;
                float _PhaseNoise;
                float _AmbientFloor;
            CBUFFER_END

            Varyings vert(Attributes input)
            {
                Varyings output = (Varyings)0;
                output.positionCS = TransformObjectToHClip(input.positionOS.xyz);
                output.color = input.color * _Color;
                output.uv = TRANSFORM_TEX(input.uv, _MainTex);
                output.worldPos = TransformObjectToWorld(input.positionOS.xyz).xy;
                output.screenUV = ComputeScreenPos(output.positionCS);
                return output;
            }

            half4 frag(Varyings input) : SV_Target
            {
                float2 uv = input.uv;

                float2 cellPos = floor(input.worldPos);
                float hash = frac(sin(dot(cellPos, float2(12.9898, 78.233))) * 43758.5453);
                float phase = hash * _PhaseNoise * 6.2831853;

                float dx = sin(_Time.y * _FlowSpeed + uv.y * _FlowFrequency + phase) * _FlowAmplitude;
                float dy = sin(_Time.y * _FlowSpeed * 1.3 + uv.x * _FlowFrequency * 0.7 + phase) * _FlowAmplitude;

                half4 baseCol = SAMPLE_TEXTURE2D(_MainTex, sampler_MainTex, uv);
                float lum = dot(baseCol.rgb, float3(0.299, 0.587, 0.114));
                float darkWeight = saturate(1.0 - lum);

                float2 flowUV = uv + float2(dx, dy) * darkWeight;
                half4 flowedColor = SAMPLE_TEXTURE2D(_MainTex, sampler_MainTex, flowUV) * input.color;

                #if USE_SHAPE_LIGHT_TYPE_0
                float2 lightUV = input.screenUV.xy / input.screenUV.w;
                half4 light = SAMPLE_TEXTURE2D(_ShapeLightTexture0, sampler_ShapeLightTexture0, lightUV);
                #else
                half4 light = half4(1, 1, 1, 1);
                #endif

                light.rgb = max(light.rgb, half3(_AmbientFloor, _AmbientFloor, _AmbientFloor));

                half3 finalRgb = flowedColor.rgb * light.rgb;
                finalRgb *= flowedColor.a;

                return half4(finalRgb, flowedColor.a);
            }
            ENDHLSL
        }
    }
}
```
