### 一、5 行对齐摘要

1. **根本目标**：为基础电格提供立体的“赛博萤火虫/微型球状闪电”常驻环绕粒子，随地块状态自动创建与回收，告别无美术贴图的单调纯色方块。
2. **关键假设**：单格常驻 3~5 颗青黄发光微粒，粒子利用 `Orbital Z` 切向速度绕中心盘旋；处于 `Actor` 图层（Order 330），不遮挡角色真身。
3. **受影响物理路径**：`TileVfxDirector.cs`、`TileElectricityVfxTool.cs`、`Vfx_Tile_Electricity.prefab`、测试文件。
4. **验证方式**：在场景中生成基础火池/电源，观察电格中心是否有微小光球盘旋；水球砸入转为导电区时特效是否自然延续，地貌消失时是否即刻回池。
5. **核心风险**：重开场景或切关时未清空活动粒子导致残留漂浮（已在 `ClearAll` 与 `Detach` 中严格加锁防御）。

#### 产物 1：地貌常驻特效调度器 `TileVfxDirector.cs`
- **物理相对路径**：`Assets/Scripts/Presentation/Grid/TileVfxDirector.cs`
- **设计意图**：遵循与 `TileOverlayDirector` 完全同构的架构。监听 `TileStateChanged` 事件，维护电格粒子池（`maxSize: 32`），借还时严格执行 `Clear/Play` 与 `StopEmittingAndClear`，杜绝池化残留闪烁。

```csharp
using System.Collections.Generic;
using DeepseaOil.Foundation;
using DeepseaOil.Logic.Events;
using DeepseaOil.Logic.Grid;
using DeepseaOil.Presentation.Adapters;
using UnityEngine;
using cfg.dso;

namespace DeepseaOil.Presentation.Grid
{
    /// <summary>地貌常驻特效管理器：管理电格微型球状闪电等状态绑定的环绕粒子</summary>
    /// <remarks>借还池对象严格执行 StopEmittingAndClear，避免池化复用时的残留粒子暴闪。</remarks>
    [DisallowMultipleComponent]
    public sealed class TileVfxDirector : MonoBehaviour
    {
        private const string DefaultPrefabPath = "effects/Vfx_Tile_Electricity";

        [Header("池容量")]
        [Tooltip("常驻特效对象上限，超出则丢弃新请求")]
        [SerializeField] private int maxPoolSize = 32;

        [Header("预制体引用（留空则从 Resources 懒加载）")]
        [SerializeField] private GameObject electricityVfxPrefab = default;

        [Header("接线（留空自动抓取）")]
        [SerializeField] private TilemapAdapter tilemapAdapter = default;

        private readonly Dictionary<Vector3Int, GameObject> _activeVfx = new();
        private readonly HashSet<TileStateType> _electricStates = new()
        {
            TileStateType.BasicElectricity,
            TileStateType.ConductZone,
            TileStateType.ChargedThorn,
        };

        private Pool<GameObject> _pool;
        private GridGeometry _geometry;
        private bool _attached;

        public int ActiveVfxCount => _activeVfx.Count;

        public void Initialize(in GridGeometry geometry)
        {
            if (_pool != null) return;

            _geometry = geometry;

            _pool = new Pool<GameObject>(
                factory: CreateVfxObject,
                onGet: OnGetFromPool,
                onRelease: OnReleaseToPool,
                name: "TileVfxPool",
                maxSize: maxPoolSize,
                overflowPolicy: PoolOverflowPolicy.CreateOrDrop,
                onDestroy: Destroy);
        }

        public void Attach()
        {
            if (_attached) return;

            if (tilemapAdapter == null) tilemapAdapter = GetComponent<TilemapAdapter>() ?? FindObjectOfType<TilemapAdapter>();
            if (tilemapAdapter != null) Initialize(tilemapAdapter.ReadGeometry());

            EventBus<TileStateChanged>.Subscribe(OnTileStateChanged);
            _attached = true;
        }

        public void Detach()
        {
            if (!_attached) return;

            EventBus<TileStateChanged>.Unsubscribe(OnTileStateChanged);
            _attached = false;
            ClearAll();
        }

        public void ClearAll()
        {
            if (_pool == null) return;

            foreach (KeyValuePair<Vector3Int, GameObject> kv in _activeVfx)
            {
                if (kv.Value != null) _pool.Release(kv.Value);
            }

            _activeVfx.Clear();
        }

        private void Start()
        {
            Attach();
        }

        private void OnDestroy()
        {
            Detach();
            _pool?.Dispose();
            _pool = null;
        }

        private void OnTileStateChanged(TileStateChanged evt)
        {
            if (_pool == null) return;

            // 1. 若旧状态有常驻特效，先行归还
            if (_activeVfx.TryGetValue(evt.Cell, out GameObject oldGo))
            {
                _pool.Release(oldGo);
                _activeVfx.Remove(evt.Cell);
            }

            // 2. 判定新状态是否属于带电地貌
            if (!_electricStates.Contains(evt.State)) return;
            if (!_geometry.IsValid) return;

            if (!_pool.TryGet(out GameObject vfxObj) || vfxObj == null) return;

            Vector2 center = _geometry.CellCenter(evt.Cell);
            vfxObj.transform.position = new Vector3(center.x, center.y, 0f);

            _activeVfx[evt.Cell] = vfxObj;
        }

        private GameObject CreateVfxObject()
        {
            GameObject prefab = electricityVfxPrefab;
            if (prefab == null)
            {
                prefab = Resources.Load<GameObject>(DefaultPrefabPath);
            }

            GameObject go;
            if (prefab != null)
            {
                go = Instantiate(prefab, transform);
            }
            else
            {
                go = new GameObject("Vfx_Tile_Electricity_Fallback");
                go.transform.SetParent(transform, false);
                go.AddComponent<ParticleSystem>();
            }

            go.SetActive(false);
            return go;
        }

        private static void OnGetFromPool(GameObject go)
        {
            go.SetActive(true);
            ParticleSystem[] systems = go.GetComponentsInChildren<ParticleSystem>(true);
            for (int i = 0; i < systems.Length; i++)
            {
                systems[i].Clear(true);
                systems[i].Play(true);
            }
        }

        private static void OnReleaseToPool(GameObject go)
        {
            ParticleSystem[] systems = go.GetComponentsInChildren<ParticleSystem>(true);
            for (int i = 0; i < systems.Length; i++)
            {
                systems[i].Stop(true, ParticleSystemStopBehavior.StopEmittingAndClear);
            }
            go.SetActive(false);
        }
    }
}
```

#### 产物 2：Editor 预制体全自动生成与装配工具 `TileElectricityVfxTool.cs`
- **物理相对路径**：`Assets/Editor/Effects/TileElectricityVfxTool.cs`
- **设计意图**：**杜绝人工在 Inspector 漏填 ParticleSystem 复杂参数**。点击菜单一键生成配置完美的 `Vfx_Tile_Electricity.prefab`（注入切向环绕速度、钟形渐变曲线、Unlit 材质、Actor 330 层级），并自愈挂载到主场景的 `TilemapAdapter`。

```csharp
#if UNITY_EDITOR
using System.IO;
using DeepseaOil.Presentation.Adapters;
using DeepseaOil.Presentation.Grid;
using DeepseaOil.Presentation.Visual;
using UnityEditor;
using UnityEngine;

namespace DeepseaOil.EditorTools.Effects
{
    public static class TileElectricityVfxTool
    {
        private const string MenuPath = "DeepseaOil/特效/一键生成电格萤火虫预制体与接线";
        private const string PrefabDir = "Assets/Resources/effects";
        private const string PrefabPath = "Assets/Resources/effects/Vfx_Tile_Electricity.prefab";

        [MenuItem(MenuPath)]
        public static void GeneratePrefabAndWireScene()
        {
            if (!Directory.Exists(PrefabDir)) Directory.CreateDirectory(PrefabDir);

            GameObject vfxPrefab = EnsureVfxPrefab();
            WireToScene(vfxPrefab);

            AssetDatabase.SaveAssets();
            Debug.Log("[VFX] 电格萤火虫特效已生成完毕：预制体与场景接线 100% 闭环就绪。");
        }

        private static GameObject EnsureVfxPrefab()
        {
            GameObject tempGo = new GameObject("Vfx_Tile_Electricity");
            ParticleSystem ps = tempGo.AddComponent<ParticleSystem>();

            // 1. Main 模块配置
            ParticleSystem.MainModule main = ps.main;
            main.duration = 2.0f;
            main.loop = true;
            main.startLifetime = new ParticleSystem.MinMaxCurve(1.6f, 2.2f);
            main.startSpeed = 0.05f;
            main.startSize = new ParticleSystem.MinMaxCurve(0.12f, 0.18f);
            main.startColor = new Color(0.490f, 0.976f, 1.0f, 1.0f); // #7DF9FF 青蓝电色
            main.simulationSpace = ParticleSystemSimulationSpace.Local;
            main.playOnAwake = false;
            main.stopAction = ParticleSystemStopAction.None;

            // 2. Emission 模块配置（极致克制：常驻维持 3~4 颗微粒）
            ParticleSystem.EmissionModule emission = ps.emission;
            emission.rateOverTime = 3.5f;

            // 3. Shape 模块配置（半格圆域发射）
            ParticleSystem.ShapeModule shape = ps.shape;
            shape.shapeType = ParticleSystemShapeType.Circle;
            shape.radius = 0.35f;
            shape.arc = 360f;

            // 4. Velocity over Lifetime 模块（核心切向环绕盘旋）
            ParticleSystem.VelocityOverLifetimeModule vel = ps.velocityOverLifetime;
            vel.enabled = true;
            vel.orbitalZ = new ParticleSystem.MinMaxCurve(2.8f, 3.6f); // 产生立交环绕
            vel.radial = -0.15f; // 微向心力防甩出格子

            // 5. Color over Lifetime 模块（青白与亮黄交替渐变）
            ParticleSystem.ColorOverLifetimeModule colorMod = ps.colorOverLifetime;
            colorMod.enabled = true;
            Gradient grad = new Gradient();
            grad.SetKeys(
                new[]
                {
                    new GradientColorKey(new Color(0.490f, 0.976f, 1.0f), 0.0f),
                    new GradientColorKey(new Color(0.976f, 0.808f, 0.353f), 0.5f), // 混入明黄
                    new GradientColorKey(new Color(0.490f, 0.976f, 1.0f), 1.0f)
                },
                new[]
                {
                    new GradientAlphaKey(0.0f, 0.0f),
                    new GradientAlphaKey(1.0f, 0.2f),
                    new GradientAlphaKey(1.0f, 0.8f),
                    new GradientAlphaKey(0.0f, 1.0f)
                }
            );
            colorMod.color = grad;

            // 6. Size over Lifetime 模块（钟形平滑过渡）
            ParticleSystem.SizeOverLifetimeModule sizeMod = ps.sizeOverLifetime;
            sizeMod.enabled = true;
            AnimationCurve curve = new AnimationCurve();
            curve.AddKey(0f, 0.2f);
            curve.AddKey(0.5f, 1.0f);
            curve.AddKey(1.0f, 0.2f);
            sizeMod.size = new ParticleSystem.MinMaxCurve(1.0f, curve);

            // 7. Renderer 模块配置（Unlit 材质，Actor 图层，Order 330）
            ParticleSystemRenderer psr = tempGo.GetComponent<ParticleSystemRenderer>();
            Shader unlitShader = Shader.Find("Universal Render Pipeline/Particles/Unlit");
            if (unlitShader != null)
            {
                psr.sharedMaterial = new Material(unlitShader);
                AssetDatabase.CreateAsset(psr.sharedMaterial, "Assets/Resources/effects/M_TileElectricity_Particle.mat");
            }

            psr.sortingLayerName = SortingLayerConsts.Actor;
            psr.sortingOrder = 330; // 高于贴地 310/320，低于站立实体 400

            GameObject prefab = PrefabUtility.SaveAsPrefabAsset(tempGo, PrefabPath);
            Object.DestroyImmediate(tempGo);
            return prefab;
        }

        private static void WireToScene(GameObject vfxPrefab)
        {
            TilemapAdapter adapter = Object.FindObjectOfType<TilemapAdapter>();
            if (adapter == null) return;

            TileVfxDirector director = adapter.GetComponent<TileVfxDirector>();
            if (director == null) director = adapter.gameObject.AddComponent<TileVfxDirector>();

            SerializedObject so = new SerializedObject(director);
            so.FindProperty("electricityVfxPrefab").objectReferenceValue = vfxPrefab;
            so.FindProperty("tilemapAdapter").objectReferenceValue = adapter;
            so.ApplyModifiedProperties();

            EditorUtility.SetDirty(adapter.gameObject);
        }
    }
}
#endif
```
