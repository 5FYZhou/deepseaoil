#### `TileStateMachine.cs`（新）

```csharp
namespace DeepseaOil.Logic.Grid
{
    /// <summary>格子状态机。只管切换与回调，不管 Tick 调度。</summary>
    public sealed class TileStateMachine
    {
        private readonly Dictionary<TileStateId, ITileState> _states = new();
        private ITileState _current;

        public TileStateId CurrentId => _current?.Id ?? default;

        public void Register(ITileState state) { _states[state.Id] = state; }

        public void SwitchTo(TileStateId id, in TileContext ctx)
        {
            if (_current?.Id.Equals(id) == true) return;   // 同状态不切
            _current?.OnExit(in ctx);
            _current = _states.TryGetValue(id, out var s) ? s : null;
            _current?.OnEnter(in ctx);
        }

        public void Tick(in TileContext ctx) => _current?.OnTick(in ctx);
    }
}
```

#### `TileTickQueue.cs`（新）

```csharp
namespace DeepseaOil.Logic.Grid
{
    /// <summary>双缓冲 Tick 队列。本帧提交的请求下一帧才处理，防止同帧递归。</summary>
    /// <remarks>
    /// 一次性消费：处理完即从队列移除。状态想周期 Tick 必须在 OnTick 里重新提交。
    /// 延迟由状态自己管理（用 ctx.Now / ctx.DeltaTime），队列只负责"哪些格下帧要 Tick"。
    /// </remarks>
    public sealed class TileTickQueue
    {
        private readonly List<Vector3Int> _current = new();
        private readonly List<Vector3Int> _next = new();
        private readonly HashSet<Vector3Int> _dedup = new();  // 同帧去重

        public void Schedule(Vector3Int cell)
        {
            if (_dedup.Add(cell)) _next.Add(cell);
        }

        public void Swap()
        {
            (_current, _next) = (_next, _current);
            _next.Clear();
            _dedup.Clear();
        }

        public IReadOnlyList<Vector3Int> Current => _current;
    }
}
```

#### `IEnemyCellRegistry.cs` / `EnemyCellRegistry.cs`（新）

```csharp
namespace DeepseaOil.Logic.Grid
{
    /// <summary>敌人 → 格子 的归属映射。单点判定（脚底中心）。</summary>
    public interface IEnemyCellRegistry
    {
        void Register(EnemyId id, Vector3Int cell);
        void Unregister(EnemyId id);
        void Move(EnemyId id, Vector3Int newCell);
        IReadOnlyList<EnemyId> GetIn(Vector3Int cell);
    }

    public sealed class EnemyCellRegistry : IEnemyCellRegistry
    {
        private readonly Dictionary<Vector3Int, List<EnemyId>> _cellToEnemies = new();
        private readonly Dictionary<EnemyId, Vector3Int> _enemyToCell = new();
        private static readonly List<EnemyId> Empty = new();

        public void Register(EnemyId id, Vector3Int cell)
        {
            _enemyToCell[id] = cell;
            if (!_cellToEnemies.TryGetValue(cell, out var list))
                _cellToEnemies[cell] = list = new List<EnemyId>();
            list.Add(id);
        }

        public void Unregister(EnemyId id)
        {
            if (!_enemyToCell.TryGetValue(id, out var cell)) return;
            _enemyToCell.Remove(id);
            if (_cellToEnemies.TryGetValue(cell, out var list))
            {
                list.Remove(id);
                if (list.Count == 0) _cellToEnemies.Remove(cell);
            }
        }

        public void Move(EnemyId id, Vector3Int newCell)
        {
            if (_enemyToCell.TryGetValue(id, out var old) && old == newCell) return;
            Unregister(id);
            Register(id, newCell);
        }

        public IReadOnlyList<EnemyId> GetIn(Vector3Int cell)
            => _cellToEnemies.TryGetValue(cell, out var list) ? list : Empty;
    }
}
```

#### `GridLogic.cs`（新，核心门面）

```csharp
namespace DeepseaOil.Logic.Grid
{
    /// <summary>格子系统的逻辑入口。持有全部格子状态、调度 Tick、处理球落地与联动。</summary>
    public sealed class GridLogic : ITileScheduler
    {
        private readonly Dictionary<Vector3Int, TileEntry> _tiles = new();
        private readonly TileTickQueue _queue = new();
        private readonly EnemyCellRegistry _registry = new();
        private readonly IEnemyResolver _resolver;
        private readonly Dictionary<TileStateId, ITileState> _statePrototypes = new();

        // 每帧入口
        public void Tick(float now, float dt)
        {
            _queue.Swap();
            foreach (var cell in _queue.Current)
            {
                if (!_tiles.TryGetValue(cell, out var e)) continue;
                e.Machine.Tick(BuildContext(cell, now, dt));
            }
        }

        // 球落地：触发状态转换
        public void OnBallHit(Vector3Int cell, int ballTypeId)
        {
            if (!_tiles.TryGetValue(cell, out var e)) return;
            var next = ResolveNextState(e.Machine.CurrentId, ballTypeId);
            e.Machine.SwitchTo(next, BuildContext(cell, ...));

            // 状态转换 = 一次基础伤害（全局设定）
            DealBaseDamage(cell);
        }

        // ITileScheduler 实现
        public void ScheduleTick(Vector3Int cell) => _queue.Schedule(cell);

        // 伤害结算
        private void DealBaseDamage(Vector3Int cell)
        {
            var enemies = _registry.GetIn(cell);
            foreach (var id in enemies)
            {
                if (!_resolver.TryGet(id, out var target)) continue;
                target.TakeDamage(new Damage(cellCenter, BaseAmount, DamageSource.Tile));
            }
        }

        // 内部

        private TileStateId ResolveNextState(TileStateId cur, int ballTypeId) { /* ... */ }
        private sealed class TileEntry { public TileStateMachine Machine; public float Elapsed; }
    }
```

```csharp
namespace DeepseaOil.Logic.Grid.States
{
    /// <summary>泥浆格：进入时造成一次基础伤害，之后减速直到超时。</summary>
    public sealed class MudTileState : ITileState
    {
        public TileStateId Id { get; }
        private float _elapsed;
        private readonly float _duration;

        public void OnEnter(in TileContext ctx)
        {
            _elapsed = 0f;
            // 立刻 Tick 一次（触发首次结算）
            ctx.Scheduler.ScheduleTick(ctx.Cell);
        }

        public void OnTick(in TileContext ctx)
        {
            _elapsed += ctx.DeltaTime;

            // 对脚下敌人结算减速（伤害已在状态转换时给过）
            DealSlow(ctx);

            if (_elapsed < _duration)
                ctx.Scheduler.ScheduleTick(ctx.Cell);   // 继续 Tick
            // 否则自然结束（不再提交，状态由外部或自身逻辑切换）
        }

        public void OnExit(in TileContext ctx) { /* 清理视觉在 Presentation 层处理 */ }
    }
}
```

```csharp
namespace DeepseaOil.Presentation.Actor

    public sealed class EnemyActor : MonoBehaviour, IDamageable
    {
        private int _hp;
        private bool _dead;
        private EnemyId _id;
        private EnemyLogic _logic;
        private EnemyMotor _motor;
        private SpriteRenderer _body;
        private TextMesh _hpText;
        private IEnemyCellRegistry _registry;
        private Vector3Int _lastCell;

        public bool IsDead => _dead;
        public EnemyId Id => _id;

        public void Initialize(EnemyId id, EnemyConfig cfg, Transform target, ...)
        {
            _id = id;
            /* 建刚体、碰撞体、视效 */
            _registry.Register(_id, WorldToCell(transform.position));
        }

        private void Update()
        {
            var cell = WorldToCell(transform.position);
            if (cell != _lastCell) { _registry.Move(_id, cell); _lastCell = cell; }
        }

        // IDamageable
        public void TakeDamage(in Damage damage)
        {
            if (_dead) return;
            _hp = Mathf.Max(0, _hp - Mathf.RoundToInt(damage.Amount));
            if (damage.HasKnockback) _logic.ApplyKnockback(damage.Impulse, damage.Direction);
            if (_hp <= 0) { Die(damage.Direction); return; }
            UpdateHpText();
            UpdateBodyColor();
        }

        private void OnDestroy() => _registry.Unregister(_id);
    }
}
```

```csharp
            _aim.SetColor(_resources.WaterBallCount > 0 ? White : Red);

            }
```

```csharp
    /// <summary>球落地结算。球不再直接伤害敌人，只通知格子 + 对无生命刚体冲量。</summary>
    public sealed class LandingResolver : MonoBehaviour
    {
        [SerializeField] private GridView _gridView;

        public void Resolve(Vector2 point, BallType type)
        {
            var cell = _gridView.WorldToCell(point);

            // 1. 通知格子（走 Logic 层）
            GridLogicAccessor.Instance.OnBallHit(cell, (int)type);

            if (type == BallType.Earth)
                PushBodies(point);

            // 3. 视觉：落地闪圈走 EffectModule
            EffectModule.Play(EffectId.LandingRing, EffectContext.At(point));
        }
```
